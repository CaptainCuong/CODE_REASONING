#include <stdio.h>
#include <stdlib.h>

int fibonacci(int n) {
    if (n <= 0) return 0;
    if (n == 1) return 1;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main(int argc, char *argv[]) {
    if (argc != 2) {
        printf("Usage: %s <k>\n", argv[0]);
        return 1;
    }
    int k = atoi(argv[1]);
    printf("%d\n", fibonacci(k));
    return 0;
}

// Since we define RUST in Rust, but not in C, the C main will be used.
// That's fine.
