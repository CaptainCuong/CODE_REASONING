"cat > extract_domains.py << 'EOF'\nimport datasets\nfrom transformers import AutoTokenizer\n\n# Load dataset\ndf = datasets.load_dataset('open-thoughts/OpenThoughts-114k')\ntrain = df['train']\n\n# Function to get domain from a text\ndef get_domain(text):\n    # Look for keywords\n    keywords = ['science', 'mathematics', 'engineering']\n    lower = text.lower()\n    for kw in keywords:\n        if kw in lower:\n            return kw\n    return None\n\n# Collect all entries with their assigned domain\nentries = []\nfor entry in train:\n    sys = entry['system']\n    conv = entry['conversations']\n    # Combine system and conversations text\n    combined = sys + ' ' + ' '.join(str(conv[0]['value']) for conv in conv)\n    domain = get_domain(combined)\n    entries.append((entry, domain))\n    if domain:\n        print(f'Domain found: {domain}')\n\n# Count occurrences\nfrom collections import Counter\ndomain_counts = Counter([d for _, d in entries if d])\nprint('Domain counts:', domain_counts)\n\n# Filter science entries\nscience_entries = [entry for entry, domain in entries if domain == 'science']\nprint(f'Number of science entries: {len(science_entries)}')\n\n# Tokenize with Qwen2.5-1.5B-Instruct\ntokenizer = AutoTokenizer.from_pretrained('Qwen2.5-1.5B-Instruct')\ntotal_tokens = sum(len(tokenizer.tokenize(entry['text'])) for entry in science_entries)\nprint(f'Total tokens: {total_tokens}')\n\n# Write to answer.txt\nwith open('/app/answer.txt', 'w') as f:\n    f.write(str(total_tokens))\nEOF\n"
