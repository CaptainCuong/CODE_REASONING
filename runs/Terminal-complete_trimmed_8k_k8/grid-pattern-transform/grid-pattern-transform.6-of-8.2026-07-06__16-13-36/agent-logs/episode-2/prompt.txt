def solve(input_grid):
    """
    Transform the input grid according to the pattern.

    Args:
        input_grid (List[List[int]]): 2D list of integers representing the input grid
            0: Black     1: Blue       2: Red        3: Green     4: Yellow
            5: Grey      6: Hot Pink   7: Orange     8: Light Blue 9: Maroon

    Returns:
        List[List[int]]: 2D list of integers representing the transformed grid
    """
    # Extract rows
    row0 = input_grid[0]
    row1 = input_grid[1]

    # Build the 6x6 output grid
    output = []
    # Row 0: row0 repeated 3 times
    output.append(row0 * 3)
    # Row 1: row1 repeated 3 times
    output.append(row1 * 3)
    # Row 2: [row1[0], row0[0]] repeated 3 times
    output.append([row1[0], row0[0]] * 3)
    # Row 3: [row1[1], row0[1]] repeated 3 times
    output.append([row1[1], row0[1]] * 3)
    # Row 4: row0 repeated 3 times
    output.append(row0 * 3)
    # Row 5: row1 repeated 3 times
    output.append(row1 * 3)

    return output
