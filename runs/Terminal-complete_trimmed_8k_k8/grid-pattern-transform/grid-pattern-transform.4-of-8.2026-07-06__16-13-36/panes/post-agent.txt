import numpy as np

def solve(input_grid):
    """
    Transform the input grid according to the pattern.

    Args:
        input_grid (List[List[int]]): 2D list of integers representing the input grid
            0: Black     1: Blue       2: Red        3: Green     4: Yellow
            5: Grey      6: Hot Pink   7: Orange     8: Light Blue 9: Maroon

    Returns:
        List[List[int]]: 2D list of integers representing the transformed grid
    """
    # Get grid dimensions
    height = len(input_grid)
    width = len(input_grid[0])

    # Extract the two values from the input grid (assuming 2x2)
    val0 = input_grid[0][0]
    val1 = input_grid[0][1]

    # Create output grid with 6x6 dimensions
    output_grid = []
    for row_index in range(6):
        if row_index < 2:
            # First two rows: repeat input row twice
            row = [val0, val1, val0, val1, val0, val1]
        elif row_index < 4:
            # Next two rows: alternating values three times
            row = [val1, val0, val1, val0, val1, val0]
        else:
            # Last two rows: same as first two rows
            row = [val0, val1, val0, val1, val0, val1]
        output_grid.append(row)

    return output_grid

def solve(input_grid):
    """
    Transform the input grid according to the pattern.

    Args:
        input_grid (List[List[int]]): 2D list of integers representing the input grid
            0: Black     1: Blue       2: Red        3: Green     4: Yellow
            5: Grey      6: Hot Pink   7: Orange     8: Light Blue 9: Maroon

    Returns:
        List[List[int]]: 2D list of integers representing the transformed grid
    """
    # Extract the four values from the input grid
    val0 = input_grid[0][0]
    val1 = input_grid[0][1]
    val2 = input_grid[1][0]
    val3 = input_grid[1][1]

    # Build output grid
    output = []
    output.append([val0, val1, val0, val1, val0, val1])   # Row 0
    output.append([val2, val3, val2, val3, val2, val3])   # Row 1
    output.append([val1, val0, val1, val0, val1, val0])   # Row 2 (swap first row values)
    output.append([val3, val2, val3, val2, val3, val2])   # Row 3 (swap second row values)
    output.append([val0, val1, val0, val1, val0, val1])   # Row 4
    output.append([val2, val3, val2, val3, val2, val3])   # Row 5

    return output
