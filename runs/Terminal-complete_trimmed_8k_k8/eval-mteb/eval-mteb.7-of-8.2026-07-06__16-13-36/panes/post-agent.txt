>
> # Load model

>
> # Function to compute embedding
> def embed(texts):
>     inputs = tokenizer(texts, padding=True, truncation=True, return_tensors='pt')
>     outputs = model(inputs['input_ids'], attention_mask=inputs['attention_mask'])
>     return outputs['pooler_output'].numpy()
>
> # Compute on dataset
> times = mteb.utils.Times()
> with times:
>     dataset = dataset.load_data()
>     X = dataset['text1']
>     X2 = dataset['text2']
>     y = dataset['score']
>     X_enc = embed(X)
>     X2_enc = embed(X2)
>     scores = (X_enc * X2_enc).sum(axis=1)
>
> print('Scores computed')
>
> # Prepare results
> results = {
>     'task_name': 'STS12',
>     'dataset_name': 'sts12',
>     'model_name': model_name,
>     'commit': commit,
>     'organization': org,
>     'results': {
>         'scores': scores.tolist(),
>         'times': times.get_dict()
>     }
