import torch
import torch.nn as nn


class MnistModel(nn.Module):
    def __init__(
        self,
        input_dim: int,
        hidden_dim: int,
        num_classes: int,
    ) -> torch.nn.Module:
        super().__init__()

        self.input_layer = nn.Linear(input_dim, hidden_dim)
        self.hidden_layer = nn.Linear(hidden_dim, hidden_dim)
        self.output_layer = nn.Linear(hidden_dim, num_classes)
        self.activation = nn.ReLU()

    def forward(self, x):
        x = self.input_layer(x)
        x = self.activation(x)

        x = self.hidden_layer(x)
        x = self.activation(x)

        x = self.output_layer(x)
        return x

> #!/usr/bin/env python3
> import sys
> import json
> import lodepng
> import numpy as np
>
> def load_weights():
>     with open('weights.json', 'r') as f:
>         data = json.load(f)
>     # Expected keys: input_layer.weight, input_layer.bias, hidden_layer.weight, hidden_layer.bias, output_layer.weight, output_layer.bias
>     # Convert to numpy arrays
>     weight_dict = {}
>     for key, val in data.items():
>         weight_dict[key] = np.array(val)
>     return weight_dict
>
> def load_image(filename):
>     with open(filename, 'rb') as f:
>         image_data, width, height = lodepng.load(f)
>     # image_data is a list of bytes (0-255)
>     img = np.array(image_data, dtype=np.uint8)
>     # Normalize to 0-1
>     img = img / 255.0
>     return img
>
> def inference(weights, image):
>     # image shape (28,28)
>     input_layer_weight = weights['input_layer.weight']
>     input_layer_bias = weights['input_layer.bias']
>     hidden_layer_weight = weights['hidden_layer.weight']
>     hidden_layer_bias = weights['hidden_layer.bias']
>     output_layer_weight = weights['output_layer.weight']
>     output_layer_bias = weights['output_layer.bias']
>
>     # Flatten image
>     input_size = 28 * 28
>     x = image.flatten()
>
>     # Input layer
>     z1 = np.dot(x, input_layer_weight) + input_layer_bias
>     a1 = np.maximum(z1, 0)
>
>     # Hidden layer
>     z2 = np.dot(a1, hidden_layer_weight) + hidden_layer_bias
>     a2 = np.maximum(z2, 0)
>
>     # Output layer
>     z3 = np.dot(a2, output_layer_weight) + output_layer_bias
>     a3 = np.exp(z3) / np.sum(np.exp(z3))
>
>     # Predict
>     predicted = np.argmax(a3)
>     return predicted
>
> def main():
>     if len(sys.argv) != 3:
>         print("Usage: cli_tool weights.json image.png")
>         sys.exit(1)
>     weights = load_weights()
>     image = load_image(sys.argv[2])
>     predicted = inference(weights, image)
>     print(predicted)
>     with open('prediction.txt', 'w') as f:
>         f.write(str(predicted))
>
> if __name__ == '__main__':
>     main()
