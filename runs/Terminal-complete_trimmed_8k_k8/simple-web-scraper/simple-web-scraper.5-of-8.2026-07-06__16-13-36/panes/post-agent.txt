>         print(f'Failed to fetch page: {response.status_code}')

>     soup = BeautifulSoup(response.text, 'html.parser')
>     books = []
>     for book in soup.find_all('div', class_='book'):
>         title = book.find('div', class_='book-title').text.strip()
>         author = book.find('div', class_='book-author').text.strip()
>         # Extract year, rating, price from spans
>         year_span = book.find('span', class_='book-year')
>         rating_span = book.find('span', class_='book-rating')
>         price_span = book.find('span', class_='book-price')
>         year = year_span.text.strip() if year_span else ''
>         rating = rating_span.text.strip() if rating_span else ''
>         price = price_span.text.strip() if price_span else ''
>         # Remove '$' and other non-numeric for price
>         if price.startswith('$'):
>             price = price[1:]
>         # Keep as strings for now
>         books.append({
>             'title': title,
>             'author': author,
>             'year': year,
>             'rating': rating,
>             'price': price
>         })
>     # Convert to DataFrame
>     df = pd.DataFrame(books, columns=['title', 'author', 'year', 'rating', 'price'])
>     # Save CSV
>     df.to_csv('/app/books.csv', index=False)
>     # Compute summary
>     total_books = len(df)
>     avg_price = df['price'].astype(float).mean() if total_books > 0 else 0
>     # Highest rated book (rating as float)
>     if 'rating' in df.columns and df['rating'].dtype in [float, int]:
>         highest_idx = df['rating'].idxmax()
>         highest_rating = df.loc[highest_idx, 'rating']
>         highest_title = df.loc[highest_idx, 'title']
>     else:
>         highest_rating = 'N/A'
>         highest_title = 'N/A'
>     # Write report
>     with open('/app/report.txt', 'w') as f:
>         f.write(f'Total books scraped: {total_books}\n')
>         f.write(f'Average book price: ${avg_price:.2f}\n')
>         f.write(f'Highest rated book: {highest_title} ({highest_rating}/5)\n')
>     print('Scraping complete.')
>
> if __name__ == '__main__':
>     main()
> EOF
