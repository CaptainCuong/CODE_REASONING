import os
import shutil

c4_sample_dir = 'c4_sample'
c4_reshard_dir = 'c4_reshard'
max_files = 30
max_size = 15 * 1024 * 1024

# Ensure c4_reshard directory exists
os.makedirs(c4_reshard_dir, exist_ok=True)

# Get list of all files in c4_sample
all_files = []
for root, dirs, files in os.walk(c4_sample_dir):
    for f in files:
        full_path = os.path.join(root, f)
        all_files.append(full_path)

print(f'Total files: {len(all_files)}')

# Create subdirectories and assign files in groups of max_files
dirs = []
for idx in range(0, len(all_files), max_files):
    sub_dir = os.path.join(c4_reshard_dir, f'subdir_{len(dirs)}')
    os.makedirs(sub_dir, exist_ok=True)
    dirs.append(sub_dir)
    for fpath in all_files[idx:idx+max_files]:
        size = os.path.getsize(fpath)
        if size > max_size:
            print(f'Warning: file {fpath} size {size} exceeds 15MB')
        shutil.copy2(fpath, os.path.join(sub_dir, os.path.basename(fpath)))

print(f'Created {len(dirs)} subdirectories')
print(f'Total files in c4_reshard: {len(all_files)}')
EOF
