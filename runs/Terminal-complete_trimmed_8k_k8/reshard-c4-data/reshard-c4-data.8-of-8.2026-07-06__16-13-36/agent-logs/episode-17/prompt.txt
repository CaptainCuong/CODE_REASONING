print(f'Total files: {len(all_files)}')

# Create subdirectories and assign files in groups of max_files
dirs = []
for idx in range(0, len(all_files), max_files):

    os.makedirs(sub_dir, exist_ok=True)
    dirs.append(sub_dir)
    for fpath in all_files[idx:idx+max_files]:
        size = os.path.getsize(fpath)
        if size > max_size:

        shutil.copy2(fpath, os.path.join(sub_dir, os.path.basename(fpath)))
