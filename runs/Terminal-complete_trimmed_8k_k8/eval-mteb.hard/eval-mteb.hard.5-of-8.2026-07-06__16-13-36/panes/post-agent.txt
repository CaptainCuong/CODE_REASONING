> def evaluate_model(model_name, hf_model_org, hf_model_name):
>     """Evaluate a model on STS12 and return results dict."""
>     dataset = STS12()
>     pairs = dataset.get_pairs()
>
>     # sentences are concatenated
>     sentences = [pair[0] + ' ' + pair[1] for pair in pairs]
>     model = SentenceTransformer(model_name)
>     sentences = model.tokenize(sentences)
>
>     # similarity scores
>     scores = [pair[2] for pair in pairs]
>
>     # Compute cosine_spearman: average cosine similarity between each embedded vector and the score vector
>     # We'll use the sklearn cosine_spearman which computes the cosine similarity between the two matrices.
>     # We'll pass sentences as matrix and scores as a column vector.
>     scores_matrix = np.array(scores).reshape(-1, 1)
>     cos_sim_matrix = cosine_spearman(sentences, scores_matrix)
>     # The matrix is of shape (n_samples, 1) because scores_matrix has one column.
>     # The cosine similarity for each pair is in the first column.
>     cos_sim = cos_sim_matrix[:, 0]
>     score = cos_sim.mean()
>
>     # Get model commit ID (approximate)
>     commit_id = model_name.split('-')[-1][:8]
>
>     result = {
>         "model": model_name,
>         "task": "STS12",
>         "score": float(score),
>         "commit_id": commit_id,
>         "hf_model_org": hf_model_org,
>         "hf_model_name": hf_model_name
>     }
