```python
import mteb
from mteb.datasets import STS12
from mteb.evaluations import CosineSpearman
import huggingface_hub
import json
import os

models_to_test = [
    'bert-base-uncased',
    'roberta-base',
    'xlm-r-bert-base',
    'deberta-base',
    'distilbert-base-uncased',
]

dataset = STS12()
evaluator = CosineSpearman()

for model_name in models_to_test:
    try:
        commit = huggingface_hub.get_repo_commit(model_name)
        commit = commit['commit']
    except Exception as e:
        print(f"Error getting commit for {model_name}: {e}")
        continue

    try:
        # Load the model
        if model_name == 'bert-base-uncased':
            from transformers import BertModel
            model = BertModel.from_pretrained(model_name)
        elif model_name == 'roberta-base':
            from transformers import RobertaModel
            model = RobertaModel.from_pretrained(model_name)
        elif model_name == 'xlm-r-bert-base':
            from transformers import XLMRModel
            model = XLMRModel.from_pretrained(model_name)
        elif model_name == 'deberta-base':
            from transformers import DebertaModel
            model = DebertaModel.from_pretrained(model_name)
        elif model_name == 'distilbert-base-uncased':
            from transformers import DistilBertModel
            model = DistilBertModel.from_pretrained(model_name)
        else:
            print(f"Model {model_name} not recognized")
            continue

        # Evaluate
        result = evaluator.compute(model, dataset)
        score = result['scores'][0]

        print(f"{model_name}: commit {commit[:10]}, score {score:.4f}")

        if score > 0.7:
            print(f"Found model with score > 0.7!")
            # Build result path
            org, name = model_name.split('-')
            result_path = f'results/{org}_{name}/{commit[:10]}/sts12.json'
            # Create directory
            os.makedirs(os.path.dirname(result_path), exist_ok=True)
            # Write result
            with open(result_path, 'w') as f:
                json.dump({'score': score, 'model': model_name}, f)
            print(f"Result saved to {result_path}")
            break
    except Exception as e:
        print(f"Error evaluating {model_name}: {e}")
        continue

if not any(score > 0.7 for model_name in models_to_test):
    print("No model achieved score > 0.7")
```
