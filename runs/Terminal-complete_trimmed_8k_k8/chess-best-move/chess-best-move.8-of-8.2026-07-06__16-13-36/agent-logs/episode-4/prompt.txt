import PIL.Image
import sys

img = PIL.Image.open('chess_puzzle.png')
print('Image size:', img.size)
# Convert to grayscale
gray = img.convert('L')
# Threshold: 128 is the middle of black (0) and white (255)
threshold = 128
binary = gray.point(lambda p: 255 if p > threshold else 0)
# Now find vertical lines: look for columns where the average pixel value changes
width, height = binary.size
v_lines = []
for col in range(width):
    avg = sum(binary.getpixel((col, row)) for row in range(height)) / height
    # If column is mostly black (0) then we have a line
    if avg < 100:  # black
        v_lines.append(col)
# Similarly horizontal lines
h_lines = []
for row in range(height):
    avg = sum(binary.getpixel((col, row)) for col in range(width)) / width
    if avg < 100:
        h_lines.append(row)
print('Vertical lines:', v_lines)
print('Horizontal lines:', h_lines)
# If we have at least 8 lines, we can try to assign squares
if len(v_lines) >= 8 and len(h_lines) >= 8:
    v_lines = sorted(v_lines)
    h_lines = sorted(h_lines)
    w = v_lines[1] - v_lines[0]
    h = h_lines[1] - h_lines[0]
    print('Square width:', w, 'height:', h)
    # Map each square
    squares = []
    for i in range(8):
        for j in range(8):
            x = v_lines[j] + w * i
            y = h_lines[i] + h * j
            # Get the pixel value (0 or 255)
            val = binary.getpixel((x, y))
            squares.append((i, j, val == 0))
    # Print the map
    for i in range(8):
        row = []
        for j in range(8):
            val = squares[j][2]
            if val:
                row.append('x')
            else:
                row.append('.')
        print(row)
