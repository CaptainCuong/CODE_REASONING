from PIL import Image
img = Image.open('chess_puzzle.png')
width, height = img.size
spacing = 80
white = (240, 217, 181)
black = (181, 136, 99)
white_set = set()
black_set = set()
for x in range(width):
    for y in range(height):
        r, g, b = img.getpixel((x, y))
        if (r, g, b) == white:
            white_set.add((x, y))
        elif (r, g, b) == black:
            black_set.add((x, y))
squares = {}
for x, y in white_set:
    c = x // spacing
    rank = 8 - (y // spacing)
    squares[(c, rank)] = 'w'
for x, y in black_set:
    c = x // spacing
    rank = 8 - (y // spacing)
    squares[(c, rank)] = 'b'
files = ['a','b','c','d','e','f','g','h']
print('Position:')
for rank in range(1, 9):
    line = ''
    for file_idx in range(1, 9):
        if (file_idx-1, rank) in squares:
            line += squares[(file_idx-1, rank)]
        else:
            line += '.'
    print(f'{rank}: {line}')
# Count material
material = {'w': {'p':0, 'n':0, 'b':0, 'r':0, 'q':0, 'k':0}, 'b': {'p':0, 'n':0, 'b':0, 'r':0, 'q':0, 'k':0}}
for (c, rank), piece in squares.items():
    f = files[c % 8]
    # piece is either 'w' or 'b'
    material[piece][f] += 1
print('\nMaterial count:')
for color in ['w', 'b']:
    print(f'{color}:', material[color])
