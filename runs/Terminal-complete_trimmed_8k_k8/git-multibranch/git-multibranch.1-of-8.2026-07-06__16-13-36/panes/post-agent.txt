#!/bin/bash
while read oldrev newrev refname
    do
        branch=${refname##*/}
        case "$branch" in
            main)
                src=/git/project/main
                dst=/var/www/main
                ;;
            dev)
                src=/git/project/dev
                dst=/var/www/dev
                ;;
            *)
                echo "Skipping $branch"
                continue
                ;;
        esac
        cp -r "$src" "$dst"
        echo "Deployed $branch to $dst"
    done
