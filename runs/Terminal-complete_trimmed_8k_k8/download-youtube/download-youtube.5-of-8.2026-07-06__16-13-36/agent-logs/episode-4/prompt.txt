# Download as mp4
video.streams.get_by_resolution('1080').download(output_path='video.mp4')

# Trim the last 10 seconds
clip = VideoFileClip('video.mp4')
# Get total duration
clip_duration = clip.duration
# If duration is less than 10 seconds, trim to whole clip
if clip_duration >= 10:
    trimmed = clip.subclip(clip_duration - 10, clip_duration)
else:
    trimmed = clip

# Save as result.mp4
trimmed.write_videofile('result.mp4', audio=False)
print('Done')
