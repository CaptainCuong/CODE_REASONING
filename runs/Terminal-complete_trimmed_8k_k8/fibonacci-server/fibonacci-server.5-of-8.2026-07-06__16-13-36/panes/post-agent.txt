> import http.server
> import urllib.parse

> import sys

> class FibonacciHandler(http.server.BaseHTTPRequestHandler):
>     def do_GET(self):
>         # Parse the path and query string
>         path = self.path
>         if '?' in path:
>             path, query = path.split('?', 1)
>             query = '?' + query
>         else:
>             query = ''
>
>         # Extract n from query string
>         params = urllib.parse.parse_qs(query, keep_blank_values=True)
>         n_str = params.get('n', [None])[0]
>
>         if n_str is None:
>             self.send_error(400, 'Missing query parameter n')
>             return
>
>         try:
>             n = int(n_str)
>         except ValueError:
>             self.send_error(400, 'Query parameter n must be an integer')
>             return
>
>         # Compute Fibonacci number
>         # We'll compute iteratively for large n up to 10000
>         # Fibonacci sequence: F(0)=0, F(1)=1
>         if n == 0:
>             fib = 0
>         elif n == 1:
>             fib = 1
>         else:
>             a, b = 0, 1
>             for _ in range(2, n + 1):
>                 a, b = b, a + b
>             fib = b
>
>         # Prepare response
>         response = {'result': fib}
>         self.send_response(200)
>         self.send_header('Content-Type', 'application/json')
>         self.end_headers()
>         self.wfile.write(json.dumps(response).encode('utf-8'))
>
> if __name__ == '__main__':
>     server = http.server.HTTPServer(('0.0.0.0', 3000), FibonacciHandler)
>     print('Server started on port 3000')
>     server.serve_forever()
> EOF
