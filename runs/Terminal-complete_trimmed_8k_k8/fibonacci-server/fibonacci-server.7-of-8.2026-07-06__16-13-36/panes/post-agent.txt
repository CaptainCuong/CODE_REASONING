import http.server
import json
import sys
from urllib.parse import urlparse, parse_qs

class FibonacciHandler(http.server.BaseHTTPRequestHandler):
    def do_GET(self):
        parsed = urlparse(self.path)
        query = parse_qs(parsed.query)
        n_str = query.get('n')
        if n_str is None:
            self.send_error(400, 'Missing query parameter n')
            return
        try:
            n = int(n_str[0])
        except ValueError:
            self.send_error(400, 'n must be an integer')
            return
        # Compute Fibonacci
        if n == 0:
            result = 0
        elif n == 1:
            result = 1
        else:
            a, b = 0, 1
            for _ in range(2, n+1):
                a, b = b, a + b
            result = b
        response = {'result': result}
        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        self.end_headers()
        self.wfile.write(json.dumps(response).encode())

if __name__ == '__main__':
    server = http.server.HTTPServer(('0.0.0.0', 3000), FibonacciHandler)
    print('Starting server on port 3000...')
    server.serve_forever()

import http.server
import json
import sys
import socket
from urllib.parse import urlparse, parse_qs

class FibonacciHandler(http.server.BaseHTTPRequestHandler):
    def do_GET(self):
        try:
            parsed = urlparse(self.path)
            query = parse_qs(parsed.query)
            n_str = query.get('n')
            if n_str is None:
                self.send_error(400, 'Missing query parameter n')
                return
            try:
                n = int(n_str[0])
            except ValueError:
                self.send_error(400, 'n must be an integer')
                return
            # Compute Fibonacci
            if n == 0:
                result = 0
            elif n == 1:
                result = 1
            else:
                a, b = 0, 1
                for _ in range(2, n+1):
                    a, b = b, a + b
                result = b
            response = {'result': result}
            self.send_response(200)
            self.send_header('Content-Type', 'application/json')
            self.end_headers()
            self.wfile.write(json.dumps(response).encode())
        except Exception as e:
            print(f'Error in handler: {e}', file=sys.stderr)
            self.send_error(500, 'Internal server error')

if __name__ == '__main__':
    server = http.server.HTTPServer(('0.0.0.0', 3000), FibonacciHandler)
    print('Starting server on port 3000...')
    print('Server socket:', server.server_address)
    try:
        server.serve_forever()
    except Exception as e:
        print(f'Server failed: {e}', file=sys.stderr)
        sys.exit(1)
