import http.server
import json
import sys
import urllib.parse

class FibonacciHandler(http.server.SimpleHTTPRequestHandler):
    def do_GET(self):
        # Extract query string
        query = self.path.split('?', 1)[1] if '?' in self.path else ''
        if not query:
            self.send_error(400, 'Missing query parameter n')
            return
        # Parse query parameters
        params = urllib.parse.parse_qs(query)
        if 'n' not in params:
            self.send_error(400, 'Missing query parameter n')
            return
        n_str = params['n'][0]
        try:
            n = int(n_str)
        except ValueError:
            self.send_error(400, 'Parameter n must be an integer')
            return
        # Compute Fibonacci number
        if n == 0:
            result = 0
        elif n == 1:
            result = 1
        else:
            a, b = 0, 1
            for _ in range(2, n+1):
                a, b = b, a + b
            result = b
        # Return JSON
        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        self.end_headers()
        self.wfile.write(json.dumps({'result': result}).encode('utf-8'))

if __name__ == '__main__':
    server = http.server.HTTPServer(('0.0.0.0', 3000), FibonacciHandler)
    print('Starting server on port 3000...')
    server.serve_forever()

        if not query:
            self.send_error(400, 'Missing query parameter n')
            return
        # Parse query parameters
        params = urllib.parse.parse_qs(query)
        if 'n' not in params:
            self.send_error(400, 'Missing query parameter n')
            return
        n_str = params['n'][0]
        try:
            n = int(n_str)
        except ValueError:
            self.send_error(400, 'Parameter n must be an integer')
            return
        # Compute Fibonacci number
        if n == 0:
            result = 0
        elif n == 1:
            result = 1
        else:
            a, b = 0, 1
            for _ in range(2, n+1):
                a, b = b, a + b
            result = b
        # Return JSON
        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        self.end_headers()
        self.wfile.write(json.dumps({'result': result}).encode('utf-8'))

if __name__ == '__main__':
    server = http.server.HTTPServer(('0.0.0.0', 3000), FibonacciHandler)
