import urllib.request
import urllib.parse
import json

def test_valid():
    url = 'http://localhost:3000/fib?n=5'
    with urllib.request.urlopen(url) as resp:
        data = resp.read()
        print('Valid n=5:')
        print(data.decode('utf-8'))
        expected = {'result': 5}
        result = json.loads(data.decode('utf-8'))
        assert result == expected, f'Expected {expected}, got {result}'
        print('PASS')

def test_missing_n():
    url = 'http://localhost:3000/fib'
    with urllib.request.urlopen(url) as resp:
        print('Missing n:')
        print('Status:', resp.code)
        assert resp.code == 400
        print('PASS')

def test_non_integer():
    url = 'http://localhost:3000/fib?n=abc'
    with urllib.request.urlopen(url) as resp:
        print('Non-integer n:')
        print('Status:', resp.code)
        assert resp.code == 400
        print('PASS')

if __name__ == '__main__':
    test_valid()
