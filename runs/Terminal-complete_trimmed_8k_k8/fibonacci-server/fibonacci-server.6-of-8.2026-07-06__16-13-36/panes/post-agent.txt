> import http.server
> import json
> import urllib.parse
> from http.server import HTTPServer, BaseHTTPRequestHandler

> class FibonacciHandler(BaseHTTPRequestHandler):
>     def do_GET(self):
>         # Parse query string
>         query = self.path.split('?', 1)[1] if '?' in self.path else ''
>         if not query:
>             self.send_error(400, 'Query parameter missing')
>             return
>
>         try:
>             params = urllib.parse.parse_qs(query)
>             n_str = params['n'][0]
>             n = int(n_str)
>         except (ValueError, KeyError):
>             self.send_error(400, 'Invalid integer n')
>             return
>
>         # Compute Fibonacci
>         if n < 0:
>             self.send_error(400, 'n must be non-negative')
>             return
>
>         # Compute nth Fibonacci (0-indexed)
>         # F(0) = 0, F(1) = 1
>         if n == 0:
>             result = 0
>         elif n == 1:
>             result = 1
>         else:
>             a, b = 0, 1
>             for _ in range(2, n+1):
>                 a, b = b, a + b
>             result = b
>
>         self.send_response(200)
>         self.send_header('Content-Type', 'application/json')
>         self.end_headers()
>         self.wfile.write(json.dumps({'result': result}).encode())
>
> if __name__ == '__main__':
>     server = HTTPServer(('0.0.0.0', 3000), FibonacciHandler)
>     print('Server starting on port 3000')
