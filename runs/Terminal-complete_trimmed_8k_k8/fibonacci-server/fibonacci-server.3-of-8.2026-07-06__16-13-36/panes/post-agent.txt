import sys
import json
import re
from http.server import HTTPServer, BaseHTTPRequestHandler

def fibonacci(n):
    if n < 0:
        return None
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

class FibonacciHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        # Check query string
        query = self.path.split('?')[1] if '?' in self.path else None
        if not query:
            self.send_error(400, 'Missing query parameter n')
            return
        # Parse n
        # Use regex to extract n value
        match = re.search(r'n=([0-9]+)', query)
        if not match:
            self.send_error(400, 'Parameter n not found or not integer')
            return
        n_str = match.group(1)
        try:
            n = int(n_str)
        except ValueError:
            self.send_error(400, 'Parameter n must be an integer')
            return
        result = fibonacci(n)
        if result is None:
            self.send_error(400, 'n must be non-negative')
            return
        # Return JSON
        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        self.end_headers()
        self.wfile.write(json.dumps({'result': result}).encode())

if __name__ == '__main__':
    server = HTTPServer(('0.0.0.0', 3000), FibonacciHandler)
    print('Starting server on port 3000...')
    server.serve_forever()
