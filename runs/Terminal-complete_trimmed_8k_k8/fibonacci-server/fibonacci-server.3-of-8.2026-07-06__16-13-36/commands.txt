"cat > server.py << 'EOF'\nimport sys\nimport json\nimport re\nfrom http.server import HTTPServer, BaseHTTPRequestHandler\n\ndef fibonacci(n):\n    if n < 0:\n        return None\n    a, b = 0, 1\n    for _ in range(n):\n        a, b = b, a + b\n    return a\n\nclass FibonacciHandler(BaseHTTPRequestHandler):\n    def do_GET(self):\n        # Check query string\n        query = self.path.split('?')[1] if '?' in self.path else None\n        if not query:\n            self.send_error(400, 'Missing query parameter n')\n            return\n        # Parse n\n        # Use regex to extract n value\n        match = re.search(r'n=([0-9]+)', query)\n        if not match:\n            self.send_error(400, 'Parameter n not found or not integer')\n            return\n        n_str = match.group(1)\n        try:\n            n = int(n_str)\n        except ValueError:\n            self.send_error(400, 'Parameter n must be an integer')\n            return\n        result = fibonacci(n)\n        if result is None:\n            self.send_error(400, 'n must be non-negative')\n            return\n        # Return JSON\n        self.send_response(200)\n        self.send_header('Content-Type', 'application/json')\n        self.end_headers()\n        self.wfile.write(json.dumps({'result': result}).encode())\n\nif __name__ == '__main__':\n    server = HTTPServer(('0.0.0.0', 3000), FibonacciHandler)\n    print('Starting server on port 3000...')\n    server.serve_forever()\nEOF\n"
