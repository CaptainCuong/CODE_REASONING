#!/usr/bin/env python3
import sys
import re
from collections import Counter

def main():
    if len(sys.argv) != 2:
        print("Usage: python3 text_processor.py <filename>")
        sys.exit(1)
    filename = sys.argv[1]
    try:
        with open(filename, 'r') as f:
            text = f.read()
    except FileNotFoundError:
        print(f"File {filename} not found.")
        sys.exit(1)

    # Split text into words, ignoring punctuation and case
    words = re.findall(r'\b[a-zA-Z]+\b', text.lower())
    counter = Counter(words)
