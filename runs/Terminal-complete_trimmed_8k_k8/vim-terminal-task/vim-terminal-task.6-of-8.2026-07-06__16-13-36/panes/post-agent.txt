> #!/usr/bin/env python3
> import sys
> import re
> from collections import Counter
>
> def main():
>     if len(sys.argv) != 2:
>         print("Usage: python3 text_processor.py <filename>")
>         sys.exit(1)
>     filename = sys.argv[1]
>     try:
>         with open(filename, 'r') as f:
>             text = f.read()
>     except FileNotFoundError:
>         print(f"File {filename} not found.")
>         sys.exit(1)
>     # Normalize: remove punctuation, convert to lowercase, split by whitespace
>     words = re.findall(r'\b[a-z]+\b', text.lower())
>     counts = Counter(words)
>     # Get top 3
>     top3 = counts.most_common(3)
>     for word, freq in top3:
>         print(f"{word}: {freq}")
>
> if __name__ == "__main__":
>     main()
> EOF
