#!/bin/bash
set -euo pipefail

LOG_DIR="/app/logs"
RULES_FILE="/app/rules/detection_rules.json"
REPORT_DIR="/app/reports"

# Parse JSON rules
function parse_rules() {
    jq -r '.rules[] | .id + "|" + .pattern + "|" + .severity + "|" + (.threshold|tostring) + "|" + (.timeframe|tostring)' "$RULES_FILE
}

# Parse log line for auth.log
function parse_auth_line() {
    line="$1"
    # Extract timestamp (format: Apr 10 08:12:33)
    if [[ "$line" =~ (\w{3}) (\d{2}) (\d{2}):(\d{2}):(\d{2}) ]]; then
        date="${BASH_REMATCH[1]} ${BASH_REMATCH[2]}"
        time="${BASH_REMATCH[3]}:${BASH_REMATCH[4]}:${BASH_REMATCH[5]}"
        timestamp="$date $time"
        # Extract IP: "from <IP>"
        if [[ "$line" =~ from (\S+) ]]; then
            ip="${BASH_REMATCH[1]}"
        else
            ip=""
        fi
    else
        timestamp=""
        ip=""
    fi
    echo "$timestamp|$ip|$line"
}

# Parse log line for http.log
function parse_http_line() {
    line="$1"
    # Extract timestamp: [10/Apr/2023:08:30:42 +0000]
    if [[ "$line" =~ \[(\d{2}/\w{3}/\d{4}):(\d{2}):(\d{2}):(\d{2}) ]]; then
        date="${BASH_REMATCH[1]}"
        time="${BASH_REMATCH[2]}:${BASH_REMATCH[3]}:${BASH_REMATCH[4]}"
        timestamp="$date $time"
        # Extract IP: first token after timestamp
        ip=$(echo "$line" | awk '{print $1}')
    else
        timestamp=""
        ip=""
    fi
    echo "$timestamp|$ip|$line"
}
