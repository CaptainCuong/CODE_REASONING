do\n            echo "$rule_id:$total"\n        done < "$REPORT_DIR/events.txt" | tr \'\\n\' \',\' | sed \'s/,$//\')\n        \n        # Build events array\n        events=$(cat "$REPORT_DIR/events.txt" | while IFS="|" read -r rule_id severity total ips;
