#!/bin/bash
set -e

LOG_DIR="/app/logs"
RULES_FILE="/app/rules/detection_rules.json"
OUTPUT_DIR="/app/reports"

# Ensure output directory exists
mkdir -p "$OUTPUT_DIR"

# Read JSON rules
RULES=$(jq -r '.rules[] | "\(.id):\(.pattern):\(.threshold):\(.severity)"' "$RULES_FILE")

# Initialize arrays
ALERTS=()
REPORT_EVENTS=()
STATS=()

# Timestamp function
timestamp() {
    date -u +'%Y-%m-%dT%H:%M:%SZ'
}

# Process each rule
while IFS=: read -r rule_id pattern threshold severity; do
    # Count matches in auth.log
    AUTH_COUNT=$(grep -c "$pattern" "$LOG_DIR/auth.log")
    # Count matches in http.log
    HTTP_COUNT=$(grep -c "$pattern" "$LOG_DIR/http.log")
    total_matches=$((AUTH_COUNT + HTTP_COUNT))

    # Collect matching IPs from both logs
    AUTH_IPS=$(grep "$pattern" "$LOG_DIR/auth.log" | grep -oEo '[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}' | sort -u)
    HTTP_IPS=$(grep "$pattern" "$LOG_DIR/http.log" | grep -oEo '[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}' | sort -u)
    # Combine unique IPs
    ALL_IPS=$(echo "$AUTH_IPS $HTTP_IPS" | tr ' ' '\n' | sort -u)

    # Store for statistics
    STATS+=("$rule_id:$total_matches")

    # If matches >= threshold, add to alerts
    if [ $total_matches -ge $threshold ]; then
        ALERTS+=("$rule_id:$severity:$total_matches:$ALL_IPS")
    fi

    # Always add to report events
    REPORT_EVENTS+=("$rule_id:$severity:$total_matches:$ALL_IPS")

done <<< "$RULES"
