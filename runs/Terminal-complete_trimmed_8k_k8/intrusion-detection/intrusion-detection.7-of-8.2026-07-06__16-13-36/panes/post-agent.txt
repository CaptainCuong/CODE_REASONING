> RULE_TIMEFRAME=""

>     local json_file="$1"
>     if [[ ! -f "$json_file" ]]; then
>         echo "Rules file not found: $json_file"
>         exit 1
>     fi
>     # Use jq if available, else fallback to grep/sed
>     if command -v jq >/dev/null 2>&1; then
>         jq -r '.rules[] | "\(.id) \(.pattern) \(.threshold) \(.timeframe) \(.severity)"' "$json_file" > /tmp/rules.txt
>     else
>         # Fallback: simple parsing (not robust)
>         echo "jq not installed, using simple parsing (may be inaccurate)"
>         grep -A5 '"rules"' "$json_file" | grep -E '"id|pattern|threshold|timeframe|severity"' | sed -n 'N;N;N;N;s/\n/ /p'
>     fi
> }
>

> extract_ips() {
>     local line="$1"
>     local pattern="$2"
>     # Use grep -o to find IPs; pattern may contain spaces
>     # We'll use a simple regex for IP addresses
>     grep -oEo "([0-9]{1,3}\.){3}[0-9]{1,3}" "$line" | sort -u
> }
>

>     parse_rules "$RULES_FILE"

>         rule_threshold=$(echo "$rule_line" | awk '{print $3}')
>         rule_timeframe=$(echo "$rule_line" | awk '{print $4}')
>         rule_severity=$(echo "$rule_line" | awk '{print $5}')
>
>         # Count matches in auth.log
>         auth_count=$(grep -c -E "$rule_pattern" "$AUTH_LOG")
>         # Count matches in http.log
>         http_count=$(grep -c -E "$rule_pattern" "$HTTP_LOG")
>         total_matches=$((auth_count + http_count))
>
>         # If total matches >= threshold, record
>         if [[ $total_matches -ge $rule_threshold ]]; then
>             # Collect unique IPs
>             auth_ips=$(grep -oEo "([0-9]{1,3}\.){3}[0-9]{1,3}" "$AUTH_LOG" | grep -E "$rule_pattern" | sort -u)
>             http_ips=$(grep -oEo "([0-9]{1,3}\.){3}[0-9]{1,3}" "$HTTP_LOG" | grep -E "$rule_pattern" | sort -u)
>             # Combine unique IPs
>             all_ips=$(echo "$auth_ips" "$http_ips" | sort -u)
>             # Build IPs string
>             ips_str=""
>             if [[ -n "$all_ips" ]]; then
>                 ips_str=$(echo "$all_ips" | tr '\n' ',' | sed 's/,$//')
>             fi
>             # Add to alerts array
>             alerts+=("$rule_id $rule_severity $total_matches $ips_str")
>             # For statistics
>             stats+=("$rule_id $total_matches")
>         fi
>     done < /tmp/rules.txt
>
>     # Generate alert.json
>     generate_alert_json
>     # Generate report.json
>     generate_report_json
> }
>

>         if [[ $first -eq 1 ]]; then
>             first=0
>         else
>             echo ',' >> "$ALERT_OUTPUT"
>         fi
>         # Split alert into parts
>         local id=${alert%% *}
>         local severity=${alert#* }
>         local severity=${severity%% *}
>         local matches=${alert#* }
>         local matches=${matches%% *}
>         local ips=${alert#* }
>         ips=${ips%% *}
>         echo '  {"id":"'$id'","severity":"'$severity'","matches":'$matches',"ips":"'$ips'"}' >> "$ALERT_OUTPUT"
>     done
>     echo '  ]}' >> "$ALERT_OUTPUT"
> }
>
> # Generate report.json
> generate_report_json() {
>     local timestamp=$(date -u +'%Y-%m-%dT%H:%M:%SZ')
>     echo '{"timestamp":"'$timestamp'","events":[' > "$REPORT_OUTPUT"
>     local first=1
>     for alert in "${alerts[@]}"; do
>         if [[ $first -eq 1 ]]; then
>             first=0
>         else
>             echo ',' >> "$REPORT_OUTPUT"
>         fi
>         local id=${alert%% *}
>         local severity=${alert#* }
>         local severity=${severity%% *}
>         local matches=${alert#* }
>         local matches=${matches%% *}
>         local ips=${alert#* }

> #!/bin/bash
> set -e
>
> LOG_DIR="/app/logs"
> RULES_FILE="/app/rules/detection_rules.json"
> AUTH_LOG="$LOG_DIR/auth.log"
> HTTP_LOG="$LOG_DIR/http.log"
> ALERT_OUTPUT="/app/alert.json"
> REPORT_OUTPUT="/app/report.json"
>
> # Read rules into associative arrays
> RULES=()
> RULE_ID=""
> RULE_PATTERN=""
> RULE_THRESHOLD=""
> RULE_TIMEFRAME=""
> RULE_SEVERITY=""
>
> # Function to parse JSON and store rules
> parse_rules() {
>     local json_file="$1"
>     if [[ ! -f "$json_file" ]]; then
>         echo "Rules file not found: $json_file"
>         exit 1
>     fi
>     # Use jq if available, else fallback to grep/sed
>     if command -v jq >/dev/null 2>&1; then
>         # Output id|pattern|threshold|timeframe|severity separated by '|'
>         jq -r '.rules[] | "\(.id)|\(.pattern)|\(.threshold)|\(.timeframe)|\(.severity)"' "$json_file" > /tmp/rules.txt
>     else
>         echo "jq not installed, using simple parsing (may be inaccurate)"
>         grep -A5 '"rules"' "$json_file" | grep -E '"id|pattern|threshold|timeframe|severity"' | sed -n 'N;N;N;N;s/\n/ /p'
>     fi
> }
>
> # Function to extract IPs from a line based on rule pattern
> extract_ips() {
>     local line="$1"
>     local pattern="$2"
>     grep -oEo "([0-9]{1,3}\.){3}[0-9]{1,3}" "$line" | sort -u
> }
>
> # Main detection logic
> detect() {
>     parse_rules "$RULES_FILE"
>     while IFS= read -r rule_line; do
>         # Split by '|'
>         rule_id=$(echo "$rule_line" | cut -d'|' -f1)
>         rule_pattern=$(echo "$rule_line" | cut -d'|' -f2)
>         rule_threshold=$(echo "$rule_line" | cut -d'|' -f3)
>         rule_timeframe=$(echo "$rule_line" | cut -d'|' -f4)
>         rule_severity=$(echo "$rule_line" | cut -d'|' -f5)
>         echo "Rule: $rule_id, Pattern: $rule_pattern, Threshold: $rule_threshold, Timeframe: $rule_timeframe, Severity: $rule_severity"
>         # Count matches in auth.log
>         auth_count=$(grep -c -E "$rule_pattern" "$AUTH_LOG")
>         # Count matches in http.log
>         http_count=$(grep -c -E "$rule_pattern" "$HTTP_LOG")
>         total_matches=$((auth_count + http_count))
>         echo "  Auth matches: $auth_count, Http matches: $http_count, Total: $total_matches"
>         # If total matches >= threshold, record
>         if [[ $total_matches -ge $rule_threshold ]]; then
>             # Collect unique IPs
>             auth_ips=$(grep -oEo "([0-9]{1,3}\.){3}[0-9]{1,3}" "$AUTH_LOG" | grep -E "$rule_pattern" | sort -u)
>             http_ips=$(grep -oEo "([0-9]{1,3}\.){3}[0-9]{1,3}" "$HTTP_LOG" | grep -E "$rule_pattern" | sort -u)
>             # Combine unique IPs
>             all_ips=$(echo "$auth_ips" "$http_ips" | sort -u)
>             # Build IPs string
>             ips_str=""
>             if [[ -n "$all_ips" ]]; then
>                 ips_str=$(echo "$all_ips" | tr '\n' ',' | sed 's/,$//')
>             fi
>             # Add to alerts array
>             alerts+=("$rule_id $rule_severity $total_matches $ips_str")
>             # For statistics
>             stats+=("$rule_id $total_matches")
>         fi
>     done < /tmp/rules.txt
>
>     echo "Alerts array size: ${#alerts[@]}"
>     echo "Stats array size: ${#stats[@]}"
>     # Generate alert.json
>     generate_alert_json
>     # Generate report.json
>     generate_report_json
> }
>
> # Generate alert.json
> generate_alert_json() {
>     local timestamp=$(date -u +'%Y-%m-%dT%H:%M:%SZ')
>     echo '{"timestamp":"'$timestamp'","alerts":[' > "$ALERT_OUTPUT"
>     local first=1
>     for alert in "${alerts[@]}"; do
>         if [[ $first -eq 1 ]]; then
>             first=0
>         else
>             echo ',' >> "$ALERT_OUTPUT"
>         fi
>         # Split alert into parts
>         local id=${alert%% *}
>         local severity=${alert#* }
>         local severity=${severity%% *}
>         local matches=${alert#* }
>         local matches=${matches%% *}
>         local ips=${alert#* }
>         ips=${ips%% *}
>         echo '  {"id":"'$id'","severity":"'$severity'","matches":'$matches',"ips":"'$ips'"}' >> "$ALERT_OUTPUT"
>     done
>     echo '  ]}' >> "$ALERT_OUTPUT"
> }
>
> # Generate report.json
> generate_report_json() {
>     local timestamp=$(date -u +'%Y-%m-%dT%H:%M:%SZ')
>     echo '{"timestamp":"'$timestamp'","events":[' > "$REPORT_OUTPUT"
>     local first=1
>     for alert in "${alerts[@]}"; do
>         if [[ $first -eq 1 ]]; then
>             first=0
>         else
>             echo ',' >> "$REPORT_OUTPUT"
>         fi
>         local id=${alert%% *}
>         local severity=${alert#* }
>         local severity=${severity%% *}
>         local matches=${alert#* }
>         local matches=${matches%% *}
>         local ips=${alert#* }
>         ips=${ips%% *}
>         echo '  {"id":"'$id'","severity":"'$severity'","matches":'$matches',"ips":"'$ips'"}' >> "$REPORT_OUTPUT"
>     done
>     echo '  ],"statistics":{' >> "$REPORT_OUTPUT"
>     local first_stat=1
>     for stat in "${stats[@]}"; do
>         if [[ $first_stat -eq 1 ]]; then
>             first_stat=0
>         else
>             echo ',' >> "$REPORT_OUTPUT"
>         fi
>         local rule_id=${stat%% *}
>         local count=${stat#* }
>         count=${count%% *}
>         echo '    "'$rule_id'":'$count >> "$REPORT_OUTPUT"
>     done
>     echo '  }}' >> "$REPORT_OUTPUT"
> }
>
> # Execute detection
> detect
> EOF
