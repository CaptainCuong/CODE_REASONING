#!/usr/bin/env python3
import subprocess
import sys
import collections

def run_maze(maze_id):
    proc = subprocess.Popen(
        ['./maze_game.sh', str(maze_id)],
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True
    )
    stdout = proc.stdout
    stdin = proc.stdin

    # Data structures
    visited = set()  # positions (row, col) of open cells
    wall_positions = set()  # positions (row, col) of walls
    start_pos = None
    exit_pos = None

    # We'll start at (0,0) and explore from there.
    current_pos = (0, 0)
    visited.add(current_pos)

    # Send an empty line to start the game
    stdin.write('\n')
    line = stdout.readline()
    if not line:
        print('Game did not start')
        return None

    # DFS stack: each element is (position, directions_tried)
    stack = []
    directions = ['N', 'E', 'S', 'W']

    # BFS for backtracking
    parent = {}

    # Helper to send a batch command
    def send_batch(cmd):
        stdin.write(cmd + '\n')
        responses = []
        while True:
            line = stdout.readline()
            if not line:
                break
            responses.append(line.strip())
        return responses

    # Main DFS loop
    while True:
        # Try all four directions from current_pos
        for d in directions:
            cmd = f'move {d} & N & E & S & W'
            responses = send_batch(cmd)
            for i, dd in enumerate(directions):
                if responses[i] == 'hit wall':
                    # wall position
                    row, col = current_pos
                    if dd == 'N':
                        new_row, new_col = row - 1, col
                    elif dd == 'E':
                        new_row, new_col = row, col + 1
                    elif dd == 'S':
                        new_row, new_col = row + 1, col
                    elif dd == 'W':
                        new_row, new_col = row, col - 1
                    wall_positions.add((new_row, new_col))
                elif responses[i] == 'moved':
                    # moved successfully
                    if dd == 'N':
                        new_row, new_col = row - 1, col
                    elif dd == 'E':
                        new_row, new_col = row, col + 1
                    elif dd == 'S':
                        new_row, new_col = row + 1, col
                    elif dd == 'W':
                        new_row, new_col = row, col - 1
                    new_pos = (new_row, new_col)
                    if new_pos not in visited:
                        visited.add(new_pos)
                        parent[new_pos] = current_pos
                        stack.append((new_pos, set()))
                elif responses[i] == 'reached exit':
                    # found exit
                    if dd == 'N':
                        new_row, new_col = row - 1, col
                    elif dd == 'E':
                        new_row, new_col = row, col + 1
                    elif dd == 'S':
                        new_row, new_col = row + 1, col
                    elif dd == 'W':
                        new_row, new_col = row, col - 1
                    exit_pos = (new_row, new_col)
                    break
            if exit_pos:
                break
        if exit_pos:
            break

        if not stack:
            break

        # Backtrack: pop the last position and try directions not yet tried
        pos, tried = stack.pop()
        if pos == exit_pos:
            break

        for d in directions:
            if d not in tried:
                tried.add(d)
