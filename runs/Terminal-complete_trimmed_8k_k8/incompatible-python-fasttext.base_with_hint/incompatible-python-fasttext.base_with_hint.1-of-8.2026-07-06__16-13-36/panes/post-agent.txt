   195      def predict(self, text, k=1, threshold=0.0, on_unicode_error="strict"):
   196          """
   197          Given a string, get a list of labels and a list of
   198          corresponding probabilities. k controls the number
   199          of returned labels. A choice of 5, will return the 5
   200          most probable labels. By default this returns only
   201          the most likely label and probability. threshold filters
   202          the returned labels by a threshold on probability. A
   203          choice of 0.5 will return labels with at least 0.5
   204          probability. k and threshold will be applied together to
   205          determine the returned labels.
   206
   207          This function assumes to be given
   208          a single line of text. We split words on whitespace (space,
   209          newline, tab, vertical tab) and the control characters carriage
   210          return, formfeed and the null character.

   228              )
   229
   230              return all_labels, all_probs
   231          else:

   236              else:

   477  def read_args(arg_list, arg_dict, arg_names, default_values):
   478      param_map = {
   479          "min_count": "minCount",
   480          "word_ngrams": "wordNgrams",
   481          "lr_update_rate": "lrUpdateRate",
   482          "label_prefix": "label",
   483          "pretrained_vectors": "pretrainedVectors",
   484      }
   485
   486      ret = {}
   487      manually_set_args = set()
   488      for (arg_name, arg_value) in chain(zip(arg_names, arg_list), arg_dict.items()):
   489          if arg_name in param_map:
   490              arg_name = param_map[arg_name]
   491          if arg_name not in arg_names:
   492              raise TypeError("unexpected keyword argument '%s'" % arg_name)
   493          if arg_name in ret:
   494              raise TypeError("multiple values for argument '%s'" % arg_name)
   495          ret[arg_name] = arg_value
   496          manually_set_args.add(arg_name)
   497
   498      for (arg_name, arg_value) in default_values.items():
   499          if arg_name not in ret:
   500              ret[arg_name] = arg_value
   501
   502      return (ret, manually_set_args)

   196      def predict(self, text, k=1, threshold=0.0, on_unicode_error="strict"):
   197          """
   198          Given a string, get a list of labels and a list of
   199          corresponding probabilities. k controls the number
   200          of returned labels. A choice of 5, will return the 5
   201          most probable labels. By default this returns only
   202          the most likely label and probability. threshold filters
   203          the returned labels by a threshold on probability. A
   204          choice of 0.5 will return labels with at least 0.5
   205          probability. k and threshold will be applied together to
   206          determine the returned labels.
   207
   208          This function assumes to be given
   209          a single line of text. We split words on whitespace (space,
   210          newline, tab, vertical tab) and the control characters carriage
   211          return, formfeed and the null character.
   212
   213          If the model is not supervised, this function will throw a ValueError.
   214
   215          If given a list of strings, it will return a list of results as usually
   216          received for a single line of text.
   217          """
   218
   219          def check(entry):
   220              if entry.find("\n") != -1:
   221                  raise ValueError("predict processes one line at a time (remove '\\n')")
   222              entry += "\n"
   223              return entry
   224
   225          if type(text) == list:
   226              text = [check(entry) for entry in text]
   227              all_labels, all_probs = self.f.multilinePredict(
   228                  text, k, threshold, on_unicode_error
   229              )
   230
   231              return all_labels, all_probs
   232          else:
   233              text = check(text)
   234              predictions = self.f.predict(text, k, threshold, on_unicode_error)
   235              if predictions:
   236                  probs, labels = zip(*predictions)
   237              else:
   238                  probs, labels = ([], ())
   239
   240              return labels, np.asarray(probs, copy=False)

   291      def get_line(self, text, on_unicode_error="strict"):
   292          """
   293          Split a line of text into words and labels. Labels must start with
   294          the prefix used to create the model (__label__ by default).
   295          """
   296
   297          def check(entry):
   298              if entry.find("\n") != -1:
   299                  raise ValueError("get_line processes one line at a time (remove '\\n')")
   300              entry += "\n"
   301              return entry
   302
   303          if type(text) == list:
   304              text = [check(entry) for entry in text]
   305              return self.f.multilineGetLine(text, on_unicode_error)
   306          else:
   307              text = check(text)
   308              return self.f.getLine(text, on_unicode_error)

   479      param_map = {
   480          "min_count": "minCount",
   481          "word_ngrams": "wordNgrams",
   482          "lr_update_rate": "lrUpdateRate",

   490          if arg_name in param_map:

   500          if arg_name not in ret:
