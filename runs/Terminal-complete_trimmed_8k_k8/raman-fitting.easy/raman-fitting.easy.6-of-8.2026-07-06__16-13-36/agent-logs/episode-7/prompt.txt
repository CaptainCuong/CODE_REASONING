import numpy as np
from scipy.optimize import curve_fit
import json

def lorentzian(x, x0, gamma, amplitude, offset):
    return amplitude / ((x - x0)**2 + gamma**2) + offset

# Load data
data = np.loadtxt(graphene.dat, delimiter=',', skiprows=0)
x = data[:, 0]
y = data[:, 1]

# Fit G peak (around 1580)
p0_G = [1580, 20, 1000, 0]
try:
    popt_G, pcov_G = curve_fit(lorentzian, x, y, p0=p0_G, maxfev=5000)
except Exception as e:
    print(f"G fit failed: {e}")
    popt_G = [1580, 20, 1000, 0]

# Fit 2D peak (around 2700)
p0_2D = [2700, 10, 1000, 0]
try:
    popt_2D, pcov_2D = curve_fit(lorentzian, x, y, p0=p0_2D, maxfev=5000)
except Exception as e:
    print(f"2D fit failed: {e}")
    popt_2D = [2700, 10, 1000, 0]

results = {
    "G": {
        "x0": float(popt_G[0]),
        "gamma": float(popt_G[1]),
        "amplitude": float(popt_G[2]),
        "offset": float(popt_G[3])
    },
    "2D": {
        "x0": float(popt_2D[0]),
        "gamma": float(popt_2D[1]),
        "amplitude": float(popt_2D[2]),
        "offset": float(popt_2D[3])
    }
}

with open('results.json', 'w') as f:
    json.dump(results, f, indent=2)
