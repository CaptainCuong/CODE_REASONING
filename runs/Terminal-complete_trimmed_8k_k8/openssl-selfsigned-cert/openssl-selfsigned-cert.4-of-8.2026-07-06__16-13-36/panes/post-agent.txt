>     # Parse subject

>     # We'll just print the whole subject
>     print('Subject:', subject)
>
>     # Parse dates: we can just print the dates line
>     print('Validity dates:', dates)
>
>     # Extract SHA-256 fingerprint
>     # fp_line: "sha256 Fingerprint=7F:57:D7:E2:F9:D9:C2:78:F8:64:E6:10:91:E8:9F:5E:E1:90:07:0B:51:B5:81:A4:1D:87:AB:22:D3:46:EC:11"
>     import re
>     match = re.search(r'Fingerprint=([0-9A-F:]+)', fp_line)

>     else:
>         print('Could not extract SHA-256 fingerprint')
>         sys.exit(1)
>
>     # Verify that the certificate can be used for TLS
>     # We'll create a simple SSL context to check

>     try:
>         context.load_cert_chain(cert_path, key_path)
