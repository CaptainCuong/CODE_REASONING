        # Check if files exist

        # Get the certificate object

        subject = cert.get_subject()
        common_name = subject.get('CN')

        # Get the expiration date (notAfter)
        not_after = cert.get_not_after()
        expiration_date = datetime.strptime(not_after, "%Y-%m-%d %H:%M:%S %z")

        # Current date
        now = datetime.now()

        # Verify certificate is not expired
        if expiration_date < now:
            print('ERROR: Certificate is expired')
            sys.exit(1)

        # Extract expiration date string in YYYY-MM-DD format
        expiration_str = expiration_date.strftime('%Y-%m-%d')

        # Print details
        print(f'Subject: {subject}')

        print(f'Common Name: {common_name}')
        print(f'Expiration Date: {expiration_str}')
        print('Certificate verification successful')

   if __name__ == '__main__':
