Load the PEM file to verify it's a valid certificate\n        with open(ssl_path, 'rb') as f:\n            cert_data = f.read()\n        # Try to load using OpenSSL via subprocess\n        # Use openssl x509 -x509opt no-CA-check -in server.pem -out /dev/null -noout -noenc\n        # Extract Common Name and expiration date from dates\n        # The dates output is like:\n        # Validity: Dec 31 00:00:00 2025 GMT\n        # Validity: Jan 1 00:00:00 2026 GMT\n        # We'll parse and output Common Name and expiration date in YYYY-MM-DD\n        # We'll use OpenSSL to get the Common Name\n        cmd = ['openssl', 'x509', '-in', '/app/ssl/server.crt', '-get_name']\n        result = subprocess.run(cmd, capture_output=True, text=True)\n        if result.returncode == 0:\n            # Parse dates to get expiration date\n        # We'll use OpenSSL to get the expiration date in a format we can parse\n        # Use -dates and then use Python to parse the dates\n        # We'll just output the dates as they are\n        # But requirement: expiration date in YYYY-MM-DD format\n        # We'll use OpenSSL to get the expiration date only\n        cmd = ['openssl', 'x509', '-in', '/app/ssl/server.crt', '-get-expired']\n        result = subprocess.run(cmd, capture_output=True, text=True)\n        if result.returncode == 0:\n            expired = result.stdout.strip()\n            # expired is like: Dec 31 00:00:00 2025 GMT\n            # Convert to YYYY-MM-DD\n            # We'll use datetime parsing\n            from datetime import datetime\n            try:\n                dt = datetime.strptime(expired, '%b %d %H:%M:%S %Y %Z')\n                expiration_date = dt.strftime('%Y-%m-%d')\n            except ValueError:\n                expiration_date = 'Unknown'\n        else:\n            expiration_date = 'Unknown'\n        \n        print(f'Common Name: {common_name}')\n        print(f'Expiration date: {expiration_date}')\n        print('Certificate verification successful')\n    except Exception as e:\n        print(f'Error loading certificate: {e}')\n        sys.exit(1)\n\nif __name__ == '__main__':\n    main()\nEOF\n"
