> import ssl

> def main():

>     if not os.path.exists(ssl_path):
>         print('ERROR: PEM file not found at', ssl_path)
>         sys.exit(1)

>         with open(ssl_path, 'rb') as f:
>             data = f.read()

>         # Subject
>         subject = cert.subject
>         # Common Name
>         common_name = subject.get('commonName')
>         # Expiration date
>         expiration_date = cert.not_after
>         # Convert to YYYY-MM-DD
>         expiration_date_str = expiration_date.strftime('%Y-%m-%d')

>         fp_proc = subprocess.run(['openssl', 'x509', '-in', ssl_path, '-sha256', '-fingerprint'], capture_output=True, text=True)
>         fp = fp_proc.stdout.split('=')[1].strip()
>         # Print required details
>         print('Subject:', subject)
>         print('Common Name:', common_name)
>         print('Expiration date:', expiration_date_str)
>         print('SHA-256 fingerprint:', fp)
>         print('Certificate verification successful')
>     except Exception as e:
>         print('ERROR: Failed to load or verify certificate:', e)
>         sys.exit(1)
>
> if __name__ == '__main__':
>     main()
