import struct

# image.ppm
def analyze_ppm(filename):
    with open(filename, 'rb') as f:
        header = f.read(100)
        lines = header.decode('ascii').split('\n')
        w, h = map(int, lines[1].split())
        maxval = int(lines[2])
        rest = f.read()
        pixels = list(map(int, rest.split()))
        print(f'{filename} dimensions: {w}x{h}')
        print(f'{filename} maxval: {maxval}')
        print(f'{filename} total pixels: {len(pixels)}')
        print(f'{filename} first 100 pixels: {pixels[:100]}')
        return w, h, pixels

w1, h1, p1 = analyze_ppm('image.ppm')

print('\nHigh-res downsampled by factor 4:')

down = []
for y in range(0, h2, 4):
    for x in range(0, w2, 4):

        down.append(round(avg))
print('  first 100 downsampled:', down[:100])
