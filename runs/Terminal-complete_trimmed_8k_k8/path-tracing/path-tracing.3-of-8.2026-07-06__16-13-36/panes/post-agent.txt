def read_ppm_p3(filename):
    with open(filename, 'r') as f:
        lines = f.readlines()
    header = lines[0].strip()
    if header != 'P3':
        raise ValueError('Not P3 format')
    width_height = lines[1].strip()
    width, height = map(int, width_height.split())
    max_val = int(lines[2].strip())
    if max_val != 255:
        raise ValueError('Max value not 255')
    data = b''.join(struct.pack('<BBB', r, g, b) for line in lines[3:] for r,g,b in [map(int, line.strip().split())])
    return width, height, data

    with open(filename, 'rb') as f:
        header = f.read(3)
        if header != b'P6':
            raise ValueError('Not P6 format')
        width_height = f.read(6)
        width, height = struct.unpack('<ii', width_height)
        max_val = struct.unpack('<i', f.read(4))[0]
        if max_val != 255:
            raise ValueError('Max value not 255')
        size = width * height * 3
        data = f.read(size)
    return width, height, data
