>
>             # Tokenize and predict

from flask import Flask, request, json
from transformers import AutoModelForSequenceClassification, AutoTokenizer
import sys
import torch

def main():
    app = Flask(__name__)

    # Load model and tokenizer
    model_path = '/app/model_cache/sentiment_model'
    model = AutoModelForSequenceClassification.from_pretrained(model_path)
    tokenizer = AutoTokenizer.from_pretrained(model_path)

    @app.route('/sentiment', methods=['POST'])
    def sentiment():
        try:
            data = request.get_json()
            text = data.get('text', '')
            if not text:
                return {'error': 'Missing text field'}

            # Tokenize and predict
            inputs = tokenizer(text, return_tensors='pt')
            outputs = model(**inputs)
            # The model returns log probabilities for each class (positive, negative)
            # For distilbert-base-uncased-finetuned-sst-2-english, the classes are 0: positive, 1: negative
            # We'll compute confidence by taking the max of the two probabilities
            confidence_positive = torch.exp(outputs.logits[0]).tolist()[0]
            confidence_negative = torch.exp(outputs.logits[0]).tolist()[1]

            # Determine sentiment
            sentiment = 'positive' if confidence_positive > confidence_negative else 'negative'

            return {
                'sentiment': sentiment,
                'confidence': {
                    'positive': confidence_positive,
                    'negative': confidence_negative
                }
            }
        except Exception as e:
            return {'error': str(e)}

    if __name__ == '__main__':
        app.run(host='0.0.0.0', port=5000, debug=False)

if __name__ == '__main__':
    main()
