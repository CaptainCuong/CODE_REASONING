>     try:
>         data = request.get_json()
>         if 'text' not in data:
>             return {'error': 'Missing text field'}, 400
>         text = data['text']
>         # Tokenize the text
>         inputs = tokenizer(text, return_tensors='pt')
>         # Get predictions
>         outputs = model(inputs['input_ids'], attention_mask=inputs['attention_mask'])
>         # Get the probability of positive sentiment (class 1)
>         probabilities = outputs.logits
>         prob_positive = probabilities[0][1].item()  # second class is positive
>         prob_negative = 1 - prob_positive
>         # Determine sentiment
>         if prob_positive > 0.5:
>             sentiment = 'positive'
>         else:
>             sentiment = 'negative'
>         # Prepare response
>         response = {
>             'sentiment': sentiment,
>             'confidence': {
>                 'positive': float(prob_positive),
>                 'negative': float(prob_negative)
>             }
>         }
>         return response
>     except Exception as e:
>         return {'error': str(e)}, 500
