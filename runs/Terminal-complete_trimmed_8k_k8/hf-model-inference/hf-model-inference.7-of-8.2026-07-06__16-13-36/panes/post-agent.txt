>     model.save_pretrained(model_path)

>
> @app.route('/sentiment', methods=['POST'])
> def sentiment():
>     try:
>         data = request.get_json()
>         if 'text' not in data:
>             return {'error': 'Missing text field in request'}, 400
>         text = data['text']
>         # Tokenize
>         inputs = tokenizer(text, return_tensors='pt')
>         # Predict
>         outputs = model(**inputs)
>         # Get probabilities
>         probabilities = outputs.logits
>         # Assuming the model outputs two classes: positive and negative (order may vary)
>         # We'll map to positive/negative based on the highest probability
>         pos_prob = probabilities[0].item() if probabilities[0].item() > probabilities[1].item() else probabilities[1].item()

>         total = pos_prob + neg_prob
>         if total == 0:
>             confidence = {'positive': 0.5, 'negative': 0.5}
>         else:
>             confidence = {'positive': pos_prob / total, 'negative': neg_prob / total}
>         # Determine sentiment
>         if pos_prob > neg_prob:
>             sentiment = 'positive'
>         else:
>             sentiment = 'negative'
>         # Return
>         return {
>             'sentiment': sentiment,
>             'confidence': confidence
>         }
>     except Exception as e:

>
