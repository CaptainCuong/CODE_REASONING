        from flask import Flask, request, jsonify
        import torch
        from transformers import AutoModelForSequenceClassification, AutoTokenizer

        model_name = "distilbert-base-uncased-finetuned-sst-2-english"
        model_path = "/app/model_cache/sentiment_model"

        # Load the model and tokenizer
        tokenizer = AutoTokenizer.from_pretrained(model_path)
        model = AutoModelForSequenceClassification.from_pretrained(model_path)

        app = Flask(__name__)

        @app.route("/sentiment", methods=["POST"])
        def sentiment():
            data = request.get_json()
            if "text" not in data:
                return {"error": "Missing 'text' field in request"}

            inputs = tokenizer(data["text"], return_tensors="pt")
            outputs = model(**inputs)
            probabilities = torch.softmax(outputs.logits, dim=1)

            # Get the sentiment label
            sentiment_label = "positive" if probabilities[0][1] > probabilities[0][0] else "negative"
            confidence = {
                "positive": float(probabilities[0][1]),
                "negative": float(probabilities[0][0])
            }

            return {
                "sentiment": sentiment_label,
                "confidence": confidence
            }

        if __name__ == "__main__":
            app.run(host="0.0.0.0", port=5000)
