import csv
import json
from datetime import datetime
from collections import defaultdict
import os

# Load schema (optional, but we can use it for validation)
with open('schema.json', 'r') as f:
    schema = json.load(f)

# Read departments
depts = {}
with open('data/departments.csv', 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        dept_id = row['id']
        dept = {
            'id': dept_id,
            'name': row['name'],
            'budget': float(row['budget']),
            'employees': [],
            'projects': []
        }
        depts[dept_id] = dept

# Read employees
employees = []
with open('data/employees.csv', 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        emp = {
            'id': row['id'],
            'name': row['name'],
            'position': row['position'],
            'skills': row['skills'].split(';'),
            'years_of_service': int(row['years_of_service'])
        }
        employees.append(emp)
        dept_id = row['department_id']
        depts[dept_id]['employees'].append(emp)

# Read projects
with open('data/projects.csv', 'r') as f:
    reader = csv.DictReader(f)
    for row in reader:
        project = {
            'name': row['name'],
            'status': row['status'],
            'members': row['member_ids'].split(';'),
            'deadline': row['deadline']
        }
        dept_id = row['department_id']
        depts[dept_id]['projects'].append(project)

# Build organization object
organization = {
    'name': 'Organization',  # placeholder, could be derived from data
    'founded': '2020-01-01',  # placeholder
    'departments': list(depts.values())
}

# Compute statistics
total_budget = sum(d['budget'] for d in depts.values())
average_department_budget = total_budget / len(depts)

total_employees = len(employees)

# skill distribution
skill_counts = defaultdict(int)
for emp in employees:
    for skill in emp['skills']:
        skill_counts[skill] += 1

# department sizes
dept_sizes = defaultdict(int)
for emp in employees:
    dept_sizes[emp['department_id']] += 1

# project status distribution
status_counts = defaultdict(int)
for proj in [p for d in depts.values() for p in d['projects']]:
    status_counts[proj['status']] += 1

# average years of service
average_years = sum(e['years_of_service'] for e in employees) / total_employees

# Build statistics object
statistics = {
    'averageDepartmentBudget': average_department_budget,
    'totalEmployees': total_employees,
    'skillDistribution': dict(skill_counts),
    'departmentSizes': dict(dept_sizes),
    'projectStatusDistribution': dict(status_counts),
    'averageYearsOfService': average_years
}

# Build final JSON
output = {
    'metadata': {
        'version': '1.0',
        'timestamp': datetime.utcnow().isoformat() + 'Z',
        'generator': 'CSV to JSON Processor'
    },
    'organization': organization,
    'statistics': statistics
}

# Write to file
with open('organization.json', 'w') as f:
    json.dump(output, f, indent=2)

print('organization.json created')
