Usage (from gitolite clone directory):

}

if ($to) {
    _mkdir($to);
    system("cp -RpP * $to");
    _print( "$to/VERSION", $version );

    # Replace shebangs if necessary.
    my $thisperl = $Config{perlpath};
    if ($thisperl ne '/usr/bin/perl') {

    _print( "VERSION", $version );
