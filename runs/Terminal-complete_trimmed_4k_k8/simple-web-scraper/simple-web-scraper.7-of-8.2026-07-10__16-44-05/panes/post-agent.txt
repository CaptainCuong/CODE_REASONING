>     author = book_div.find('div', class_='book-author').text
>     year = book_div.find('span', class_='book-year').text
>     rating = book_div.find('span', class_='book-rating').text
>     price = book_div.find('span', class_='book-price').text
>     # Clean up: remove 'Published:', 'Rating: /5', '$'
>     year = year.replace('Published: ', '')
>     rating = rating.replace('Rating: ', '').replace('/5', '')
>     price = price.replace('$', '')
>     books.append({
>         'title': title,
>         'author': author,
>         'year': year,
>         'rating': rating,
>         'price': price
>     })
>
> # Convert year to int, rating to float, price to float
> for b in books:
>     b['year'] = int(b['year'])
>     b['rating'] = float(b['rating'])
>     b['price'] = float(b['price'])
>
> # Save to CSV
> df = pd.DataFrame(books, columns=['title', 'author', 'year', 'rating', 'price'])
> df.to_csv('books.csv', index=False)
>
> # Summary report
> total_books = len(books)
> average_price = df['price'].mean()
> highest_rated = max(books, key=lambda x: x['rating'])
>
