> import requests
> import bs4
> import pandas as pd
> import re
>
> # Fetch the page
> url = 'http://server:8000'
> response = requests.get(url)
> response.raise_for_status()
>
> # Parse with BeautifulSoup
> soup = bs4.BeautifulSoup(response.text, 'html.parser')
>
> # Find all book elements
> books = []
> for book_div in soup.find_all('div', class_='book'):
>     title_div = book_div.find('div', class_='book-title')
>     author_div = book_div.find('div', class_='book-author')
>     details_div = book_div.find('div', class_='book-details')
>     if title_div and author_div and details_div:
>         title = title_div.get_text().strip()
>         author = author_div.get_text().strip()
>         # Extract year, rating, price from spans
>         year_span = details_div.find('span', class_='book-year')
>         rating_span = details_div.find('span', class_='book-rating')
>         price_span = details_div.find('span', class_='book-price')
>         year = year_span.get_text().strip().replace('Published: ', '')
>         # rating may be like 'Rating: 4.2/5'
>         rating_str = rating_span.get_text().strip().replace('Rating: ', '')
>         # split by '/' to get numeric rating
>         rating_num = float(rating_str.split('/')[0])
>         # price may be like '$12.99'
>         price_str = price_span.get_text().strip().replace('$', '')
>         price = float(price_str)
>         books.append({
>             'title': title,
>             'author': author,
>             'year': year,
>             'rating': rating_num,
>             'price': price
>         })
>
> # Create DataFrame
> df = pd.DataFrame(books)
> # Write CSV
> df.to_csv('books.csv', index=False, columns=['title', 'author', 'year', 'rating', 'price'])
>
> # Compute summary
> total_books = len(df)
> average_price = df['price'].mean()
> # Highest rated book (if tie, pick first)
> highest_rated = df.loc[df['rating'].idxmax()]
>
> # Write report
> with open('report.txt', 'w') as f:
>     f.write(f'Total books: {total_books}\n')
>     f.write(f'Average price: ${average_price:.2f}\n')
>     f.write(f'Highest rated book: {highest_rated["title"]} by {highest_rated["author"]} with rating {highest_rated["rating"]}\n')
>
> print('Scraping complete.')
> EOF
