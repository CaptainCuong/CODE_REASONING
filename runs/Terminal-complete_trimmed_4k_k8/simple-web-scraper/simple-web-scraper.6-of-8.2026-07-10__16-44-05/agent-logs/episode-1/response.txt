```python
import requests
from bs4 import BeautifulSoup
import pandas as pd
import sys

def main():
    url = "http://server:8000"
    response = requests.get(url)
    soup = BeautifulSoup(response.text, 'html.parser')
    
    books = []
    for book in soup.find_all('div', class_='book'):
        # Extract title
        title = book.find('div', class_='book-title').text
        # Extract author
        author = book.find('div', class_='book-author').text
        # Extract year, rating, price
        year = book.find('span', class_='book-year').text
        rating = book.find('span', class_='book-rating').text
        price = book.find('span', class_='book-price').text
        
        # Clean up: remove extra text (like "Published: " and "$")
        year = year.replace("Published: ", "").strip()
        rating = rating.replace("Rating: ", "").strip()
        price = price.replace("$", "").strip()
        
        # Convert types
        year = int(year)
        rating = float(rating.split('/')[0])  # rating is like "4.2/5"
        price = float(price)
        
        books.append({
            'title': title,
            'author': author,
            'year': year,
            'rating': rating,
            'price': price
        })
    
    # Create DataFrame
    df = pd.DataFrame(books, columns=["title", "author", "year", "rating", "price"])
    # Save to CSV
    df.to_csv('books.csv', index=False)
    
    # Summary report
    total_books = len(books)
    avg_price = df['price'].mean()
    highest_rated = df.loc[df['rating'].idxmax()]
    
    with open('report.txt', 'w') as f:
        f.write(f"Total books: {total_books}\\n")
        f.write(f"Average price: {avg_price:.2f}\\n")
        f.write(f"Highest rated book: {highest_rated['title']} (Rating: {highest_rated['rating']}/5)\\n")
    
    print("Done. Books saved to books.csv and report to report.txt")

if __name__ == "__main__":
    main()
```
