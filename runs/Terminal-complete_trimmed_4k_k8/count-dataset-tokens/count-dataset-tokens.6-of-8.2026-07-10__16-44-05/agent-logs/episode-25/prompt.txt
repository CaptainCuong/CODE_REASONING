import json
from datasets import load_dataset
from transformers import AutoTokenizer

d = load_dataset('open-thoughts/OpenThoughts-114k')
# Use the train split
total_tokens = 0
for entry in d['train']:
    conv = entry['conversations']
    domain = None
    for c in conv:
        if 'domain' in c:
            domain = c['domain']
            break
    if domain == 'science':
        for c in conv:
            text = c.get('value', '')
            tokenizer = AutoTokenizer.from_pretrained('Qwen2.5-1.5B-Instruct')
            tokens = tokenizer.tokenize(text)
            total_tokens += len(tokens)
            # print(f"{domain}: {len(tokens)} tokens for '{text}'")
print('Total tokens in science domain:', total_tokens)
# Write to answer.txt
with open('/app/answer.txt', 'w') as f:
    f.write(str(total_tokens))
EOF
