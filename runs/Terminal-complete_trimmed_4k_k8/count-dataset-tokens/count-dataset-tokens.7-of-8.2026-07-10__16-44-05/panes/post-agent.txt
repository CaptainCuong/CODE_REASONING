import os
from transformers import AutoTokenizer

dataset_path = '/app/dataset.jsonl'
total_tokens = 0

# Check if dataset is JSONL
if os.path.exists(dataset_path):
    with open(dataset_path, 'r') as f:
        for line in f:
            line = line.strip()
            if line:
                try:
                    entry = json.loads(line)
                except json.JSONDecodeError:
                    continue
                # Check if domain is science
                if entry.get('domain') == 'science':
                    # Tokenize
                    tokenizer = AutoTokenizer.from_pretrained('Qwen2.5-1.5B-Instruct')
                    tokens = tokenizer.tokenize(entry['text'])
                    total_tokens += len(tokens)
                    # Optional: print progress
                    # print(f'Processed line {i}: {len(tokens)} tokens')
    print(f'Total tokens in science domain: {total_tokens}')
else:
    print('Dataset file not found')
