import json
from transformers import Qwen2_5Tokenizer
import requests
import io

# Load dataset from huggingface
url = 'https://huggingface.co/open-thoughts/OpenThoughts-114k/raw/2024-10-15-000000/00001-00001.json'
response = requests.get(url)
data = json.loads(response.text)

# Filter by domain 'science'
science_entries = [entry for entry in data if entry.get('domain') == 'science']

# Initialize tokenizer
tokenizer = Qwen2_5Tokenizer.from_pretrained('Qwen2.5-1.5B-Instruct')

total_tokens = 0
for entry in science_entries:
    text = entry.get('text', '')
    tokens = tokenizer.tokenize(text)
    total_tokens += len(tokens)

print(f'Total tokens in science domain: {total_tokens}')
with open('/app/answer.txt', 'w') as f:
    f.write(str(total_tokens))
