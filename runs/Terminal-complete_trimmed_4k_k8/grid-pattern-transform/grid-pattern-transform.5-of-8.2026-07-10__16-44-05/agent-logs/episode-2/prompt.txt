import numpy as np

def solve(input_grid):
    """
    Transform the input grid according to the pattern.

    Args:
        input_grid (List[List[int]]): 2D list of integers representing the input grid

    Returns:
        List[List[int]]: 2D list of integers representing the transformed grid
    """
    # Get grid dimensions
    height = len(input_grid)
    width = len(input_grid[0])

    # Create output grid with same dimensions (but we know output is 6x6)
    output = [[0] * (width * 3) for _ in range(height * 3)]

    # Fill output by repeating rows and columns
    for i in range(height):
        row = input_grid[i]
        for j in range(3):
            # Repeat row three times
            for k in range(3):
                output[i * 3 + j][k * width: k * width + width] = row

    return output
