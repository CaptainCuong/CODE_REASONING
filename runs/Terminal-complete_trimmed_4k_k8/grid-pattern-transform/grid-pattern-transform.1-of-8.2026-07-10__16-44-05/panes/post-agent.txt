import numpy as np

def solve(input_grid):
    """
    Transform the input grid according to the pattern.
    """
    # Create a 6x6 output grid
    output = [[0 for _ in range(6)] for _ in range(6)]

    # For each cell in input (2x2)
    for i in range(2):
        for j in range(2):
            val = input_grid[i][j]
            # Checkerboard pattern: place val where (row + col) % 2 == i % 2
            for row in range(6):
                for col in range(6):
                    if (row + col) % 2 == i % 2:
                        output[row][col] = val
    return output

def solve(input_grid):
    """
    Transform the input grid according to the pattern.
    """
    # Create a 6x6 output grid

    # For each cell in input (2x2)

    return output
