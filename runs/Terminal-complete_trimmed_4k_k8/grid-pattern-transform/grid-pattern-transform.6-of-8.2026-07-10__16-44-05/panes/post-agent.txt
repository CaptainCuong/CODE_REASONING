import numpy as np

def solve(input_grid):
    """
    Transform the input grid according to the pattern.
    """
    # Get grid dimensions
    height = len(input_grid)
    width = len(input_grid[0])

    # Create output grid with same dimensions
    output_grid = [[0 for _ in range(width)] for _ in range(height)]

    # Pattern: repeat each row 3 times, and repeat each row again 3 times total 6 rows
    # But we need to produce 6 rows. So we can just repeat the entire input 3 times and then each row again 3 times? Actually:
    # Input has 2 rows. We want:
    # Row0, Row0, Row0
    # Row1, Row1, Row1
    # Row0, Row0, Row0
    # Row1, Row1, Row1
    # Row0, Row0, Row0
    # Row1, Row1, Row1
    # That's 6 rows. So we can create output by iterating over input rows and appending the row 3 times.
    for i in range(height):
        for j in range(3):
            for k in range(width):
                output_grid[i * 3 + j][k] = input_grid[i][k]

    return output_grid

if __name__ == '__main__':
    test_input = [[8,6],[6,4]]
    result = solve(test_input)
    print(result)
