        List[List[int]]: 2D list of integers representing the transformed grid

    """

    Returns:
        List[List[int]]: 2D list of integers representing the transformed grid
    """
    # Get grid dimensions
    height = len(input_grid)
    width = len(input_grid[0])

    # Create output grid by repeating each row 3 times
    output = []
    for row in input_grid:
        # Repeat the row 3 times
        output.append(row * 3)
    # Since we have only 2 rows, output will be 6 rows
    return output
