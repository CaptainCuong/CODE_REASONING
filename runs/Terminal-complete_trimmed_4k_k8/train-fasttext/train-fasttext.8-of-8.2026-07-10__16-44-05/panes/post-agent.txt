import json
import sys

def decode_line(line):
    """Try to decode a line as UTF-8, if fails try UTF-16, then UTF-32."""
    for enc in ('utf-8', 'utf-16', 'utf-32'):
        try:
            return line.decode(enc, errors='ignore')
        except UnicodeDecodeError:
            continue
    # If none works, return empty string
    return ''

def main():
    with open('data/train-00000-of-00001.parquet', 'rb') as f:
        lines = f.readlines()
    extracted = []
    for line in lines:
        line = line.strip()
        if not line:
            continue
        # Try to decode as UTF-8
        try:
            data = json.loads(line.decode('utf-8'))
        except UnicodeDecodeError:
            # Try other encodings
            for enc in ('utf-16', 'utf-32'):
                try:
                    data = json.loads(line.decode(enc))
                    break
                except UnicodeDecodeError:
                    continue
            else:
                # If none works, skip
                continue
        extracted.append(data.get('text', ''))
    # Write to train.txt
    with open('train.txt', 'w') as f:
        for text in extracted:
            f.write(text + '\n')
    print(f'Extracted {len(extracted)} records')
    # Print first few
    for i, t in enumerate(extracted[:5]):
        print(f'{i}: {t[:50]}...')

if __name__ == '__main__':
    main()
