import torch
import os

# Define the model

output_dir = Path("results")
output_dir.mkdir(exist_ok=True)

# Determine the output filename
# According to MTEB format: results/{{hf model org}__{hf model name}}/{model commit id}/{task name}.json
# We don't have the commit id, but we can use the model path? The task expects the commit id.
# However, the instruction says the result file should follow the default MTEB format.
# We'll use the model_id and the commit id we don't have. We'll assume the commit id is the last part of the model path? Not.
# Since we don't have the commit id, we can use the model path as the model identifier.
# Let's use the model_id and the last part of the model path.

org = "bge"
model_name = "bge-small-en-v1.5"
# For commit id, we don't have it. We'll use the model path's hash? Not.

# We'll use the model_path and replace the last '/bge' with the model_name.

# Then the output file could be: results/bge__bge-small-en-v1.5/{model_dir.name}/{task}.json
# We'll use the model_dir name as the model identifier.
model_dir = Path(model_path).parent
model_dir_name = model_dir.name  # e.g., "bge"
# We'll use the parent of the parent? Actually, the parent is the organization.
# We'll just use the organization as the directory.
# Let's construct the output path.
output_path = output_dir / f"{org}__{model_name}" / model_dir_name / f"{task}.json"
# Create parent directories
output_path.parent.mkdir(parents=True, exist_ok=True)

# Save results
import json
with open(output_path, 'w') as f:

print(f"Results saved to: {output_path}")
