cache_dir = os.path.expanduser('~/.cache/huggingface')
model_path = os.path.join(cache_dir, 'models', 'bge-small-en-v1.5', '5c38ec7c405ec4b44b94cc5a9bb96e735b38267a')
print(f'Model path: {model_path}')

# Load the model and tokenizer
tok = AutoTokenizer.from_pretrained(model_path)
model = AutoModel.from_pretrained(model_path)

# Define the STS12 task
task = MTEBTask('sts12')
results = []
for sample in task.data:
    sent1 = sample['sent1']
    sent2 = sample['sent2']
    # Compute similarity
    encoded1 = tok.encode(sent1, return_tensors='pt')
    encoded2 = tok.encode(sent2, return_tensors='pt')
    # Use the encoder to get embeddings
    emb1 = model.encoder(encoded1).mean(axis=1)
    emb2 = model.encoder(encoded2).mean(axis=1)
    score = torch.dot(emb1, emb2).item()
    results.append(score)

# Prepare the output directory
output_dir = '/app/results/bge__bge-small-en-v1.5/5c38ec7c405ec4b44b94cc5a9bb96e735b38267a'
with open(os.path.join(output_dir, 'sts12.json'), 'w') as f:
    json.dump(results, f, indent=2)

print('Results written to:', output_dir)
