int main() {
    int src = open("high_res.ppm", O_RDONLY);

        exit(1);
    }

    fstat(src, &s);
    size_t size = s.st_size;
    int dst = open("reconstructed.ppm", O_WRONLY | O_CREAT, 0777);
    if (dst < 0) {
        close(src);
        exit(1);
    }
    char buffer[8192];
    while (size > 0) {
        int n = read(src, buffer, 8192);
        if (n < 0) {
            close(src); close(dst); exit(1);
        }
        int m = write(dst, buffer, n);
        if (m < 0) {
            close(src); close(dst); exit(1);
        }
        size -= m;
    }
    close(src); close(dst);
