import subprocess
import sys

def run_game(maze_id):
    """Run the game with a pipe to stdin."""
    # Open a pipe to the game's stdin
    with open('/dev/null', 'w') as stdin:
        process = subprocess.run(['./maze_game.sh', str(maze_id)],
                               stdin=stdin,
                               stdout=subprocess.PIPE,
                               stderr=subprocess.PIPE,
                               text=True)
    # Write the command
    sys.stdin.write('move N & E & S & W\n')
    sys.stdin.flush()
    output = process.stdout.read()
    return output

if __name__ == '__main__':
    output = run_game(1)
    print('Output:', output)
