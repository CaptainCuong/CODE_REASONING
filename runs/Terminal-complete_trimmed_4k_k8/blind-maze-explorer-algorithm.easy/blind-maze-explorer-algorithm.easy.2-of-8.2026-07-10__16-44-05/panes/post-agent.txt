def run_game(maze_id):
    """Run the game with the given maze ID and return stdout and stderr."""
    cmd = ['./maze_game.sh', str(maze_id)]
    result = subprocess.run(cmd, capture_output=True, text=True)
    return result.stdout, result.stderr

def parse_response(response):
    """Parse the game response string."""
    # The response may be empty or contain multiple responses separated by '&' when batch commands were sent.
    # If there's no '&' and no 'exit', it's a single move result.
    # We'll return a list of results.
    results = []
    if '&' in response:
        parts = response.split('&')
        for p in parts:
            p = p.strip()
            if not p:
                continue
            # Expect 'moved', 'hit wall', 'reached exit'
            results.append(p)
    else:
        results.append(response)
    return results
