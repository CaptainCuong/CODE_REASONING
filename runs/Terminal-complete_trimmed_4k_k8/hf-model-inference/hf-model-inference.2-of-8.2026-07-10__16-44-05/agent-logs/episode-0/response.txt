 We'll write the Flask API:

   from flask import Flask, request
   from transformers import AutoModel, AutoTokenizer

   app = Flask(__name__)

   # Load the model and tokenizer
   model = AutoModel.from_pretrained('/app/model_cache/sentiment_model')
   tokenizer = AutoTokenizer.from_pretrained('/app/model_cache/sentiment_model')

   @app.route('/sentiment', methods=['POST'])
   def sentiment_analysis():
       try:
           data = request.json
           text = data['text']

           # Tokenize the text
           inputs = tokenizer(text, return_tensors='pt')
           outputs = model(**inputs)

           # Get the scores
           pos_score = outputs[0, 0]
           neg_score = outputs[0, 1]

           # Normalize
           total = pos_score + neg_score
           pos_score = pos_score / total
           neg_score = neg_score / total

           sentiment = "positive" if pos_score > neg_score else "negative"

           return {
               "sentiment": sentiment,
               "confidence": {
                   "positive": pos_score.item(),
                   "negative": neg_score.item()
               }
           }

       except Exception as e:
           return {
               "error": str(e)
           }

   if __name__ == '__main__':
       app.run(host='0.0.0.0', port=5000)
