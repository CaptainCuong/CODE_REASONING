import json
import sys
from transformers import AutoModel, AutoTokenizer

def main():
    # Load model and tokenizer
    model_path = '/app/model_cache/sentiment_model'
    tokenizer = AutoTokenizer.from_pretrained(model_path)
    model = AutoModel.from_pretrained(model_path)

    # Define the sentiment analysis function
    def sentiment_analysis(text):
        inputs = tokenizer(text, return_tensors='pt', padding=True)
        outputs = model(**inputs)
        # For distilbert, the output is a tensor of shape (batch_size, 2) with scores
        # We'll assume the model returns a tensor with two columns.
        # If not, we might need to adjust.
        # We'll use the first two columns as positive and negative scores.
        # If the tensor has only one column, we can assume it's a single score and the other is 0.
        scores = outputs[0].cpu().numpy()
        if scores.shape[1] == 2:
            pos = scores[:, 0]
            neg = scores[:, 1]
        else:
            # If only one column, assume it's the positive score, negative is 0
            pos = scores[:, 0]
            neg = 0.0
        # Determine sentiment (positive if pos > neg)
        sentiment = 'positive' if pos > neg else 'negative'
        return {
            'sentiment': sentiment,
            'confidence': {
                'positive': float(pos),
                'negative': float(neg)
            }
        }

    # Flask app
    from flask import Flask, request, Response
    app = Flask(__name__)

    @app.route('/sentiment', methods=['POST'])
    def sentiment_endpoint():
        try:
            data = request.get_json()
            if 'text' not in data:
                return {'error': 'Missing text field'}, 400
            result = sentiment_analysis(data['text'])
            return json.dumps(result), 200
        except Exception as e:
            return {'error': str(e)}, 500

    if __name__ == '__main__':
        app.run(host='0.0.0.0', port=5000)
