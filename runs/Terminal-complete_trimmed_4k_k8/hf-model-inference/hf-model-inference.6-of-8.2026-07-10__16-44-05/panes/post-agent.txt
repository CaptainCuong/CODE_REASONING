import sys
import subprocess
import os
import json
from transformers import AutoModelForSequenceClassification, AutoTokenizer
from flask import Flask, request, Response

def download_model():
    """Download the sentiment model if not already present."""
    model_path = '/app/model_cache/sentiment_model'
    if not os.path.exists(model_path):
        print('Downloading model...', file=sys.stderr)
        subprocess.run(['python3', '-m', 'transformers', 'download', '--no-deps', '--model-name', 'distilbert-base-uncased-finetuned-sst-2-english', '--model-ty
pe', 'seq-classification', '--save-dir', '/app/model_cache'], capture_output=True)
        # Verify download
        if not os.listdir('/app/model_cache'):
            print('Download failed or no files downloaded', file=sys.stderr)
            sys.exit(1)
    else:
        print('Model already downloaded.')

def main():
    download_model()
    # Create Flask app
    app = Flask(__name__)

    @app.route('/sentiment', methods=['POST'])
    def sentiment():
        try:
            data = request.json
            text = data.get('text')
            if text is None:
                return {'error': 'Missing text field'}, 400
            # Load model and tokenizer
            from transformers import AutoModelForSequenceClassification, AutoTokenizer
            tokenizer = AutoTokenizer.from_pretrained('/app/model_cache/sentiment_model')
            model = AutoModelForSequenceClassification.from_pretrained('/app/model_cache/sentiment_model')
            # Tokenize and predict
            inputs = tokenizer(text, return_tensors='pt')
            outputs = model(**inputs)
            # Get confidence scores
            pos = outputs.logits[0][0].item()
            neg = outputs.logits[0][1].item()
            # Normalize to [0,1]
            total = pos + neg
            pos_norm = pos / total if total != 0 else 0.5
            neg_norm = neg / total if total != 0 else 0.5
            sentiment = 'positive' if pos_norm > neg_norm else 'negative'
            result = {
                'sentiment': sentiment,
                'confidence': {
                    'positive': round(pos_norm, 4),
                    'negative': round(neg_norm, 4)
                }
            }
            return json.dumps(result), 200
        except Exception as e:
            return {'error': str(e)}, 500

    # Run the server in background using subprocess
    print('Starting server...', file=sys.stderr)
    server_process = subprocess.Popen(['python3', 'run_service2.py'], shell=False)
    server_process.wait()

if __name__ == '__main__':
    main()
