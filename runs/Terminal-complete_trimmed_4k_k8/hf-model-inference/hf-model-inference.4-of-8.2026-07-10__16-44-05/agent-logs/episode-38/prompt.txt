import sys
from flask import Flask
from transformers import AutoModelForSequenceClassification, AutoTokenizer
from pathlib import Path

app = Flask(__name__)

# Load the model
model_path = Path('/app/model_cache/sentiment_model')
model = AutoModelForSequenceClassification.from_pretrained(model_path)

# Load the tokenizer for this model
tokenizer = AutoTokenizer.from_pretrained('/app/model_cache/sentiment_model')

@app.route('/sentiment', methods=['POST'])
def sentiment_analysis():
    try:
        data = json.loads(request.data)
        text = data['text']
        # Tokenize the text
        tokens = tokenizer([text])  # returns a list of tokenizers.Encoding objects
        # Access input_ids via attribute
        input_ids = tokens[0].ids
        # Convert to tensor
        import torch
        with torch.inference_mode():
            logits = model(torch.tensor(input_ids))
        # Convert to numpy for easier handling
        import numpy as np
        probs = np.softmax(logits.logits.numpy(), axis=1)
        positive = probs[0, 0]  # first token is positive
        negative = probs[0, 1]   # second token is negative
        # Determine sentiment
        if positive > negative:
            sentiment = 'positive'
        else:
            sentiment = 'negative'
        # Return the required JSON
        return {
            'sentiment': sentiment,
            'confidence': {
                'positive': float(positive),
                'negative': float(negative)
            }
        }
    except Exception as e:
        return {
            'error': str(e)
        }, 500

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
