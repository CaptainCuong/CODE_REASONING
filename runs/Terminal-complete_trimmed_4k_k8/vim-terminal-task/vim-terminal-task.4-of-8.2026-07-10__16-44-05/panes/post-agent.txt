> #!/usr/bin/env python3
> import sys
> import re
> from collections import Counter
>
> def main():
>     if len(sys.argv) != 2:
>         print("Usage: python text_processor.py <filename>")
>         sys.exit(1)
>     filename = sys.argv[1]
>     try:
>         with open(filename, 'r') as f:
>             text = f.read()
>     except FileNotFoundError:
>         print(f"File {filename} not found.")
>         sys.exit(1)
>     # Normalize: lowercase and replace punctuation with spaces
>     # Use regex to replace any non-alphanumeric characters (except spaces) with space
>     # But we want to keep words separated by spaces.
>     # We'll split by any non-letter character, but keep letters.
>     # Simpler: replace all non-word characters with space, then split.
>     cleaned = re.sub(r'[^\w\s]', ' ', text.lower())
>     words = cleaned.split()
>     counter = Counter(words)
>     top3 = counter.most_common(3)
>     for word, freq in top3:
>         print(f"{word}: {freq}")
>
> if __name__ == '__main__':
>     main()
> EOF
