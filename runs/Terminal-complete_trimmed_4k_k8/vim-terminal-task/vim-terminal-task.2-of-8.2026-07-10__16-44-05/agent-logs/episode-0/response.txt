   def process_file(filename):

       # Remove punctuation and convert to lowercase
       words = re.findall(r'\b\w+\b', text.lower())
       counter = Counter(words)
       top3 = counter.most_common(3)
       for word, freq in top3:
           print(f"{word}: {freq}")

   if __name__ == '__main__':
