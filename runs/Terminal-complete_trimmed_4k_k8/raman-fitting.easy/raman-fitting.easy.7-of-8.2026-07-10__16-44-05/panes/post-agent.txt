import numpy as np
import scipy
from scipy import stats
import json

def fit_peak(x, y, peak_index):
    """Fit a single peak to the data."""
    # We'll use scipy's curvefit with a custom function.
    # Function: y = offset + amplitude * (gamma / ( (x - x0)^2 + gamma^2 ))
    def lorentzian(x, x0, gamma, amplitude, offset):
        return offset + amplitude * (gamma / ( (x - x0)**2 + gamma**2 ))

    # Initial guess: we can use the peak index to approximate.
    # We'll find the index of the peak.
    peak_idx = np.argmax(y)

    # Initial guess for x0
    x0 = x[peak_idx]
    # Initial guess for gamma (width) - we can use a fixed value, but better to estimate.
    # We'll set gamma to a small value and then fit.
    # Initial guess for amplitude (peak height)
    amp = y[peak_idx]
    # Initial guess for offset (average background)
    offset = np.mean(y)

    # Use scipy.curvefit to refine.
    params, _ = scipy.curvefit(lorentzian, x, y, [x0, 0.01, amp, offset], maxfev=1000)

    return {
        'x0': params[0],
        'gamma': params[1],
        'amplitude': params[2],
        'offset': params[3]
    }

# Load data
x = []
y = []
with open('/app/graphene.dat', 'r') as f:
    for line in f:
        line = line.strip()
        if line:
            parts = line.split()
            if len(parts) >= 2:
                x.append(float(parts[0]))
                y.append(float(parts[1]))

x = np.array(x)
y = np.array(y)

# Fit G peak (around 1580)
g_peak = fit_peak(x, y, 0)  # index 0 because G is lower wavenumber
# Fit 2D peak (around 2700)
two_d_peak = fit_peak(x, y, 1)  # index 1 because 2D is higher wavenumber

        'gamma': two_d_peak['gamma'],
        'amplitude': two_d_peak['amplitude'],
        'offset': two_d_peak['offset']
    }
}

import numpy as np
import scipy
import json

from scipy.optimize import minimize

def parse_line(line):
    line = line.strip()
    if not line:
        return None
    parts = line.split()
    if len(parts) != 2:
        return None
    x_str = parts[0].replace(',', '')
    y_str = parts[1].replace(',', '')
    try:
        x = float(x_str)
        y = float(y_str)
    except ValueError:
        return None
    return x, y

def load_data(filename):
    x = []
    y = []
    with open(filename, 'r') as f:
        for line in f:
            parsed = parse_line(line)
            if parsed is not None:
                x.append(parsed[0])
                y.append(parsed[1])
    return np.array(x), np.array(y)

def fit_peak(x, y, peak_pos):
    """Fit a single peak to the data."""
    # We'll use scipy's minimize to minimize the error
    # Function: y = offset + amplitude * (gamma / ( (x - x0)^2 + gamma**2 ))
    def lorentzian(x, x0, gamma, amplitude, offset):
        return offset + amplitude * (gamma / ( (x - x0)**2 + gamma**2 ))

    # Initial guess for x0
    x0 = peak_pos
    # Initial guess for gamma (width)
    gamma = 50  # reasonable value
    # Initial guess for amplitude (peak height)
    amp = y.max()
    # Initial guess for offset (average background)
    offset = np.mean(y)

    # Define the error function that takes only the parameters
    def error(params):
        x0, gamma, amp, offset = params
        y_pred = offset + amp * (gamma / ( (x - x0)**2 + gamma**2 ))
        return np.sum((y - y_pred)**2)

    # Use minimize to minimize the error
    # We need to pass bounds for gamma, amp, offset? Not required.
    # We'll set bounds to avoid negative gamma or amp.
    bounds = [(0, None), (0, None), (0, None), (0, None)]
    result = minimize(error, x0=[x0, gamma, amp, offset], bounds=bounds)

    return {
        'x0': result.x[0],
        'gamma': result.x[1],
        'amplitude': result.x[2],
        'offset': result.x[3]
    }
