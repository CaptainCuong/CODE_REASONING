import numpy as np
import scipy
import json
from scipy import stats
from scipy.optimize import minimize

def parse_line(line):
    line = line.strip()
    if not line:
        return None
    parts = line.split()
    if len(parts) != 2:
        return None
    x_str = parts[0].replace(',', '')
    y_str = parts[1].replace(',', '')
    try:
        x = float(x_str)
        y = float(y_str)
    except ValueError:
        return None
    return x, y

def load_data(filename):
    x = []
    y = []
    with open(filename, 'r') as f:
        for line in f:
            parsed = parse_line(line)
            if parsed is not None:
                x.append(parsed[0])
                y.append(parsed[1])
    return np.array(x), np.array(y)

def fit_peak(x, y, peak_pos):
    """Fit a single peak to the data."""
    # We'll use scipy's minimize to minimize the error
    # Function: y = offset + amplitude * (gamma / ( (x - x0)^2 + gamma**2 ))
    def lorentzian(x, x0, gamma, amplitude, offset):
        return offset + amplitude * (gamma / ( (x - x0)**2 + gamma**2 ))

    # Initial guess for x0
    x0 = peak_pos
    # Initial guess for gamma (width)
    gamma = 50  # reasonable value
    # Initial guess for amplitude (peak height)
    amp = y.max()
    # Initial guess for offset (average background)
    offset = np.mean(y)

    # Define the error function that takes only the parameters
    def error(params):
        x0, gamma, amp, offset = params
        y_pred = offset + amp * (gamma / ( (x - x0)**2 + gamma**2 ))
        return np.sum((y - y_pred)**2)

    # Use minimize to minimize the error
    # We need to pass bounds for gamma, amp, offset? Not required.
    # We'll set bounds to avoid negative gamma or amp.
    bounds = [(0, None), (0, None), (0, None), (0, None)]
    result = minimize(error, x0=[x0, gamma, amp, offset], bounds=bounds)

    return {
        'x0': result.x[0],
        'gamma': result.x[1],
        'amplitude': result.x[2],
        'offset': result.x[3]
    }

# Load data
x, y = load_data('/app/graphene.dat')

# Find the positions of the two peaks
peak_indices = []
for i in range(1, len(y)-1):
    if y[i] > y[i-1] and y[i] > y[i+1]:
        peak_indices.append(i)

# We expect two peaks: G around 1580 and 2D around 2700
# Let's find the indices corresponding to those ranges.
g_idx = None
two_d_idx = None
for idx in peak_indices:
    if 1580 < x[idx] < 1590:
        g_idx = idx
    elif 2700 < x[idx] < 2710:
        two_d_idx = idx

if g_idx is not None and two_d_idx is not None:
    print(f'G peak index: {g_idx}, x: {x[g_idx]:.1f}')
    print(f'2D peak index: {two_d_idx}, x: {x[two_d_idx]:.1f}')
    g_peak = fit_peak(x, y, x[g_idx])
    two_d_peak = fit_peak(x, y, x[two_d_idx])
else:
    # If not found, we can try to find the two highest y values
    sorted_indices = sorted(peak_indices, key=lambda i: y[i], reverse=True)
    g_peak = fit_peak(x, y, x[sorted_indices[0]])
    two_d_peak = fit_peak(x, y, x[sorted_indices[1]])
    print('Could not locate peaks by position, using highest y values.')

# Prepare results
results = {
    'G': {
        'x0': g_peak['x0'],
        'gamma': g_peak['gamma'],
        'amplitude': g_peak['amplitude'],
        'offset': g_peak['offset']
    },
    '2D': {
        'x0': two_d_peak['x0'],
        'gamma': two_d_peak['gamma'],
        'amplitude': two_d_peak['amplitude'],
        'offset': two_d_peak['offset']
    }
}

# Write to file
with open('/app/results.json', 'w') as f:
    json.dump(results, f, indent=2)
