    scores.append(cosine_sim(a, b))

# Compute the Pearson correlation
scores = np.array(scores)
corr = np.corrcoef(scores)[0,1]

print(f"Average cosine similarity: {scores.mean()}")
print(f"Pearson correlation: {corr}")
