import sys
import os
import subprocess
import shutil
import json
import numpy as np
import mteb
import sentence_transformers
from sentence_transformers import SentenceTransformer, Word movers
from sklearn import preprocessing
from sklearn.linear_model import LinearRegression
from sklearn.metrics import pearsonr

def main():
    # Define the model
    model_path = '/app/roberta-base-nli-stereo'
    # Ensure model exists
    if not os.path.exists(model_path):
        print(f'Model path {model_path} does not exist')
        sys.exit(1)

    # Load the STS12 dataset
    dataset = mteb.load_dataset('sts12', task_name='STS12')

    # Prepare data
    all_scores = []
    all_embeddings = []
    for sample in dataset:
        # Get sentence pairs
        s1 = sample['sentence1']
        s2 = sample['sentence2']
        # Tokenize and embed
        # We'll use the model to embed both sentences
        # The model expects a list of strings, so we create a batch
        batch = [s1, s2]
        batch_embeddings = model.encode(batch)
        # Compute similarity
        similarity = batch_embeddings[0].dot(batch_embeddings[1])
        all_scores.append(similarity)
        all_embeddings.append(batch_embeddings)

    # Compute correlation
    if len(all_scores) > 1:
        # Compute linear regression to get correlation coefficient
        X = np.array([all_scores]).T
        y = np.array([all_scores]).T
        # Add column of ones for intercept
        X_with_intercept = np.column_stack([np.ones(len(X)), X])
        reg = LinearRegression().fit(X_with_intercept, y)
        corr = reg.coef_[1]
        # Also compute Pearson correlation
        corr_pearson, _ = pearsonr(all_scores, all_scores)
        # Take the maximum
        score = max(abs(corr), abs(corr_pearson))
    else:
        score = 0.0

    print(f'StS12 score: {score}')

    # Create results directory
    os.makedirs('/app/results', exist_ok=True)
    # Create the specific directory for the model
    # We need the model's organization and name
    # We can get from the model path or from the model object
    # For simplicity, we'll use the model name as 'roberta-base-nli-stereo'
    model_name = 'roberta-base-nli-stereo'
    # The organization is 'sentence-transformers' (from sentence_transformers library)
    org = 'sentence-transformers'
    # The model commit ID is the last hash from the model path
    # We'll extract the last 8 characters as a placeholder
    model_id = '01234567'

    # Write the results
    result_path = f'/app/results/{org}__{model_name}/{model_id}/STS12.json'
    os.makedirs(os.path.dirname(result_path), exist_ok=True)

    with open(result_path, 'w') as f:
        json.dump({'score': score}, f)

    print(f'Results written to {result_path}')

if __name__ == '__main__':
    main()
