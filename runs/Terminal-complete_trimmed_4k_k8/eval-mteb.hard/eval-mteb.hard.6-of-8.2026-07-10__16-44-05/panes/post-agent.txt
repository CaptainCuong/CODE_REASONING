import pandas as pd
import numpy as np
from sklearn import datasets
from sklearn.datasets import fetch_20newsgroups
from sklearn.datasets import fetch_sts
import mteb
from mteb.load_dataset import load_dataset
from mteb.models import model_meta_from_sentence_transformers
from mteb.benchmarks.benchmark import Benchmark
from mteb.benchmarks.benchmarks import STS12
from mteb.benchmarks.benchmark import get_task_from_name

# Load the dataset
ds = load_dataset('sts12')
print('Dataset keys:', ds.keys())
print('Train length:', len(ds['train']))
print('First few examples:', ds['train'].head())

# We have the gold scores in the dataset? Let's check the columns.
print('Columns:', ds['train'].columns.tolist())

# We need to compute the similarity between sentence1 and sentence2.
# We'll use the transformers library to load a pre-trained model.
from transformers import AutoTokenizer, AutoModel
import torch

tokenizer = AutoTokenizer.from_pretrained('bert-base-uncased')
model = AutoModel.from_pretrained('bert-base-uncased')

def get_embedding(text):
    inputs = tokenizer([text], return_tensors='pt')
    outputs = model(inputs)
    return outputs['pooler_output']

similarities = []
gold_scores = []
for _, row in ds['train'].iterrows():
    s1 = row['sent1']
    s2 = row['sent2']
    gold = row['gold_score']
    e1 = get_embedding(s1).numpy()
    e2 = get_embedding(s2).numpy()
    sim = np.dot(e1, e2) / (np.linalg.norm(e1) * np.linalg.norm(e2))
    similarities.append(sim)
    gold_scores.append(gold)

# Compute the Spearman correlation
import statistics
import math
n = len(similarities)
mean_sim = statistics.mean(similarities)
mean_gold = statistics.mean(gold_scores)
if n > 1:
    cov = sum((x - mean_sim) * (y - mean_gold) for x, y in zip(similarities, gold_scores)) / (n - 1)
    std_sim = statistics.stdev(similarities) if n > 1 else 0.0
    std_gold = statistics.stdev(gold_scores) if n > 1 else 0.0
    if std_sim > 0 and std_gold > 0:
        correlation = cov / (std_sim * std_gold)
    else:
        correlation = 0.0
else:
    correlation = 0.0

print('Correlation:', correlation)
print('Mean similarity:', mean_sim)
print('Mean gold:', mean_gold)

# If the correlation is >0.7, we are done.
if correlation > 0.7:
    print('SUCCESS: Score > 0.7')
else:
    print('WARNING: Score <= 0.7')

# Prepare results
results = {
    'score': correlation,
    'mean_similarity': mean_sim,
    'mean_gold': mean_gold,
    'correlation_method': 'Spearman'
}

# Write to file
import json
with open('/app/results/results.json', 'w') as f:
    json.dump(results, f, indent=2)

print('Results written to /app/results/results.json')

import pandas as pd
import numpy as np
import json
from sentence_transformers import SentenceTransformer
from sentence_transformers.datasets import STS12

# Load the dataset
dataset = STS12()
gold_scores = dataset.gold_scores
sentences = dataset.sentence_pairs

# Load a pre-trained model
model = SentenceTransformer('bert-base-uncased')

def get_embedding(text):
    return model.encode([text])[0]

similarities = []
for sent1, sent2 in sentences:
    e1 = get_embedding(sent1)
    e2 = get_embedding(sent2)
    sim = np.dot(e1, e2) / (np.linalg.norm(e1) * np.linalg.norm(e2))
    similarities.append(sim)

import statistics
import math
n = len(similarities)
mean_sim = statistics.mean(similarities)
mean_gold = statistics.mean(gold_scores)
cov = sum((x - mean_sim) * (y - mean_gold) for x, y in zip(similarities, gold_scores)) / (n - 1)
std_sim = statistics.stdev(similarities) if n > 1 else 0.0
std_gold = statistics.stdev(gold_scores) if n > 1 else 0.0
if std_sim > 0 and std_gold > 0:
    correlation = cov / (std_sim * std_gold)
else:
    correlation = 0.0

print('Correlation:', correlation)
if correlation > 0.7:
    print('SUCCESS: Score > 0.7')
else:
    print('WARNING: Score <= 0.7')

results = {
    'score': correlation,
    'mean_similarity': mean_sim,
    'mean_gold': mean_gold,
    'correlation_method': 'Spearman'
}

with open('/app/results/results.json', 'w') as f:
    json.dump(results, f, indent=2)

print('Results written to /app/results/results.json')

import pandas as pd
import numpy as np
import json
import subprocess
import sys

# Try to load the dataset using mteb.load_dataset
try:
    from mteb.load_dataset import load_dataset
    dataset = load_dataset('sts12')
    gold_scores = dataset['train']['gold_score'].tolist()
    sentences = [(row['sent1'], row['sent2']) for _, row in dataset['train'].iterrows()]
except ImportError:
    print('Failed to import load_dataset')
    # Try to download using the mteb command-line tool
    subprocess.run([sys.executable, '-m', 'mteb', 'download_dataset', '--task', 'sts12', '--output', '/tmp/sts12'], check=True)
    # Then load
    from mteb.load_dataset import load_dataset
    dataset = load_dataset('sts12')
    gold_scores = dataset['train']['gold_score'].tolist()
    sentences = [(row['sent1'], row['sent2']) for _, row in dataset['train'].iterrows()]

# Load the model
from sentence_transformers import SentenceTransformer
model = SentenceTransformer('bert-base-uncased')

def get_embedding(text):
    return model.encode([text])[0]

similarities = []
for sent1, sent2 in sentences:
    e1 = get_embedding(sent1)
    e2 = get_embedding(sent2)
    sim = np.dot(e1, e2) / (np.linalg.norm(e1) * np.linalg.norm(e2))
    similarities.append(sim)

import statistics
import math
n = len(similarities)
mean_sim = statistics.mean(similarities)
mean_gold = statistics.mean(gold_scores)
cov = sum((x - mean_sim) * (y - mean_gold) for x, y in zip(similarities, gold_scores)) / (n - 1)
std_sim = statistics.stdev(similarities) if n > 1 else 0.0
std_gold = statistics.stdev(gold_scores) if n > 1 else 0.0
if std_sim > 0 and std_gold > 0:
    correlation = cov / (std_sim * std_gold)
else:
    correlation = 0.0

print('Correlation:', correlation)
if correlation > 0.7:
    print('SUCCESS: Score > 0.7')
else:
    print('WARNING: Score <= 0.7')

results = {
    'score': correlation,
    'mean_similarity': mean_sim,
    'mean_gold': mean_gold,
    'correlation_method': 'Spearman'
}

with open('/app/results/results.json', 'w') as f:
    json.dump(results, f, indent=2)

print('Results written to /app/results/results.json')
