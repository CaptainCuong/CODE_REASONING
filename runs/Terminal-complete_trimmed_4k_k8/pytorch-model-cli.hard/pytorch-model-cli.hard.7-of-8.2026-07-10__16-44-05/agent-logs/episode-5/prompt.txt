#!/usr/bin/env python3
import sys
import torch
import lodepng
import json
import os

def load_image(filename):
    # Load PNG file and convert to PyTorch tensor
    with open(filename, 'rb') as f:
        data = f.read()
    width, height, data, _ = lodepng.decompress(data)
    # Convert to tensor
    tensor = torch.tensor(data, dtype=torch.float32).view(1, -1)
    return tensor / 255.0  # Normalize

def main():
    if len(sys.argv) != 3:
        print("Usage: ./cli_tool weights.json image.png")
        sys.exit(1)
    weights_file = sys.argv[1]
    image_file = sys.argv[2]

    # Load model
    model = torch.load(weights_file)

    # Load image
    img = load_image(image_file)

    # Run inference
    output = model(img)
    # Find the index of the maximum value
    digit = torch.argmax(output).item()

    # Write to prediction.txt
    with open('prediction.txt', 'w') as f:
        f.write(str(digit))

    # Also print to stdout
    print(digit)

if __name__ == '__main__':
    main()
