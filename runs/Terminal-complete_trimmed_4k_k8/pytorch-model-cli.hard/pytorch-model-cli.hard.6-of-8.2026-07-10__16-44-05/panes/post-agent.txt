import torch
import torch.nn as nn


class MnistModel(nn.Module):
    def __init__(
        self,
        input_dim: int,
        hidden_dim: int,
        num_classes: int,
    ) -> torch.nn.Module:
        super().__init__()

        self.input_layer = nn.Linear(input_dim, hidden_dim)
        self.hidden_layer = nn.Linear(hidden_dim, hidden_dim)
        self.output_layer = nn.Linear(hidden_dim, num_classes)
        self.activation = nn.ReLU()

    def forward(self, x):
        x = self.input_layer(x)
        x = self.activation(x)

        x = self.hidden_layer(x)
        x = self.activation(x)

        x = self.output_layer(x)
        return x

import torch
import numpy as np
import matplotlib.pyplot as plt
import json

def main():
    # Load the model
    model = torch.load('model.pth')
    # Extract weights and biases
    weights = {}
    # We'll iterate through the model's children
    for child in model.children():
        if isinstance(child, torch.nn.Linear):
            name = child.__class__.__name__.lower() + '.' + child.weight.shape[0].replace('torch.nn.', '')
            # Actually we know the layer names from the file
            # Let's just hardcode: we have three Linear layers.
            # We'll just extract all Linear layers by iterating over the module's parameters.
            pass
    # Simpler: iterate over all parameters
    for name, param in model.named_parameters():
        weights[name] = param.cpu().numpy().tolist()

    # Write weights.json
    with open('weights.json', 'w') as f:
        json.dump(weights, f, indent=2)

    # Create a 10x10 image with digit 3
    fig, ax = plt.subplots()
    ax.set_xticks([])
    ax.set_yticks([])
    # Create a 10x10 array of zeros
    img = np.zeros((10, 10), dtype=np.uint8)
    # Draw a digit 3
    # Simple: draw lines
    # We'll use a template: 10x10 with a '3' shape.
    # We'll use a basic drawing: left vertical, top diagonal, bottom diagonal.
    # We'll create a mask.
    mask = np.zeros((10, 10), dtype=bool)
    # Left vertical line
    for i in range(10):
        mask[0, i] = True
    # Top diagonal
    for i in range(1, 8):
        mask[i, 8] = True
    # Bottom diagonal
    for i in range(2, 10):
        mask[i, 0] = True
    # Combine
    mask = mask | np.rot90(mask, 3)
    # Convert to float
    img = mask.astype(np.float32) / 255.0
    plt.imshow(img, cmap='gray')
    plt.savefig('image.png', dpi=100)
    plt.close()

    print('Generated weights.json and image.png')

if __name__ == '__main__':
    main()

#!/usr/bin/env python3
import json
import sys
import os
import math

def load_weights(filename):
    with open(filename, 'r') as f:
        return json.load(f)

def load_image(filename):
    # Simple image loader: read as bytes, convert to array of floats
    with open(filename, 'rb') as f:
        data = f.read()
    # Assuming 10x10 grayscale image, each pixel 0-255
    img = [0.0] * 100
    for i, b in enumerate(data[:100]):
        img[i] = b / 255.0
    return img

def predict(weights, img):
    # Now weights is a flat dict with keys: 'input_layer.weight', etc.
    # We need to extract the weights lists
    input_weight = weights['input_layer.weight']
    input_bias = weights['input_layer.bias']
    hidden_weight = weights['hidden_layer.weight']
    hidden_bias = weights['hidden_layer.bias']
    output_weight = weights['output_layer.weight']
    output_bias = weights['output_layer.bias']

    # Flatten the weight lists
    # input_weight: list of 250 lists (250x100) -> flatten to 25000
    input_weight_flat = [w for row in input_weight for w in row]
    hidden_weight_flat = [w for row in hidden_weight for w in row]
    output_weight_flat = [w for row in output_weight for w in row]

    # Compute hidden layer
    # hidden_layer: hidden_weight_flat length 250, input bias scalar
    hidden = 0.0
    for i in range(100):
        hidden += input_weight_flat[i] * img[i]
    hidden += input_bias[0]
    hidden += hidden_bias[0]

    # Compute output layer
    out = [0.0] * 10
    for j in range(10):
        sum_val = 0.0
        for k in range(250):
            sum_val += output_weight_flat[250*j + k] * hidden_weight_flat[k]
        out[j] = sum_val + output_bias[j]

    # Normalize
    max_val = max(out)
    return out.index(max_val)

def main():
    if len(sys.argv) != 3:
        print('Usage: ./cli_tool weights.json image.png')
        sys.exit(1)
    weights_file = sys.argv[1]
    image_file = sys.argv[2]

    weights = load_weights(weights_file)
    img = load_image(image_file)

    pred = predict(weights, img)
    print(pred)

    # Write to prediction.txt
    with open('/app/prediction.txt', 'w') as f:
        f.write(str(pred))

if __name__ == '__main__':
    main()
