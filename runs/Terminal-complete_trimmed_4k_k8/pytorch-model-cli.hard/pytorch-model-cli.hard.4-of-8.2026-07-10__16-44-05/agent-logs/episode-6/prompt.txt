import sys
import torch
import numpy as np
from PIL import Image
import json

def load_model(weights_path):
    # Try to load weights from JSON
    try:
        with open(weights_path, 'r') as f:
            weights = json.load(f)
        # Convert to torch tensor
        # Assuming weights is a dict with keys 'input_layer', 'hidden_layer', 'output_layer'
        # We'll flatten and reconstruct
        # This is a simple model; we can just load the model from the given weights
        # but we need to recompute. Let's instead load the model from model.pth
        # We'll fall back to model.pth
        # For simplicity, we'll just use the provided model.py and weights from model.pth
        # Actually we can load the model from model.pth
        return torch.load('model.pth')
    except Exception:
        pass
    # Load from model.pth
    return torch.load('model.pth')

def predict(image_path, weights_path):
    model = load_model(weights_path)
    # Convert image to tensor
    img = Image.open(image_path).convert('L')
    img_array = np.array(img).astype(float) / 255.0
    img_tensor = torch.tensor(img_array, device='cpu').view(1, -1)
    output = model(img_tensor)
    pred = torch.argmax(output).item()
    print(pred)
    return pred

def main():
    if len(sys.argv) != 3:
        print("Usage: python3 infer.py <image> <weights>")
        sys.exit(1)
    image = sys.argv[1]
    weights = sys.argv[2]
    pred = predict(image, weights)
    with open('prediction.txt', 'w') as f:
        f.write(str(pred))

if __name__ == '__main__':
    main()
