import torch
import torch.nn as nn


class MnistModel(nn.Module):
    def __init__(
        self,
        input_dim: int,
        hidden_dim: int,
        num_classes: int,
    ) -> torch.nn.Module:
        super().__init__()

        self.input_layer = nn.Linear(input_dim, hidden_dim)
        self.hidden_layer = nn.Linear(hidden_dim, hidden_dim)
        self.output_layer = nn.Linear(hidden_dim, num_classes)
        self.activation = nn.ReLU()

    def forward(self, x):
        x = self.input_layer(x)
        x = self.activation(x)

        x = self.hidden_layer(x)
        x = self.activation(x)

        x = self.output_layer(x)
        return x

import sys
import torch
import numpy as np
from PIL import Image
import json

def load_model(weights_path):
    # Try to load weights from JSON
    try:
        with open(weights_path, 'r') as f:
            weights = json.load(f)
        # Convert to torch tensor
        # Assuming weights is a dict with keys 'input_layer', 'hidden_layer', 'output_layer'
        # We'll flatten and reconstruct
        # This is a simple model; we can just load the model from the given weights
        # but we need to recompute. Let's instead load the model from model.pth
        # We'll fall back to model.pth
        # For simplicity, we'll just use the provided model.py and weights from model.pth
        # Actually we can load the model from model.pth
        return torch.load('model.pth')
    except Exception:
        pass
    # Load from model.pth
    return torch.load('model.pth')

def predict(image_path, weights_path):
    model = load_model(weights_path)
    # Convert image to tensor
    img = Image.open(image_path).convert('L')
    img_array = np.array(img).astype(float) / 255.0
    img_tensor = torch.tensor(img_array, device='cpu').view(1, -1)
    output = model(img_tensor)
    pred = torch.argmax(output).item()
    print(pred)
    return pred

def main():
    if len(sys.argv) != 3:
        print("Usage: python3 infer.py <image> <weights>")
        sys.exit(1)
    image = sys.argv[1]
    weights = sys.argv[2]
    pred = predict(image, weights)
    with open('prediction.txt', 'w') as f:
        f.write(str(pred))

if __name__ == '__main__':
    main()

import torch
import json

def extract_weights():
    model = torch.load('model.pth')
    weights = {}
    # Iterate over the model's modules
    for layer_name, layer in model.items():
        if isinstance(layer, torch.nn.Module):
            for param_name, param in layer.named_parameters():
                weights[f'{layer_name}.{param_name}'] = param.cpu().numpy().tolist()
    with open('weights.json', 'w') as f:
        json.dump(weights, f, indent=2)
    print('Weights saved to weights.json')

if __name__ == '__main__':

import sys
import torch
import numpy as np
from PIL import Image, ImageDraw
import json
import os

def load_model(weights_path):
    # Load from weights.json if exists, else from model.pth
    if os.path.exists(weights_path):
        with open(weights_path, 'r') as f:
            weights = json.load(f)
        # Reconstruct the model from weights
        model = torch.nn.Sequential()
        input_layer = torch.nn.Linear(28 * 28, 64)
        hidden_layer = torch.nn.Linear(64, 64)
        output_layer = torch.nn.Linear(64, 10)
        model.add_module('input_layer', input_layer)
        model.add_module('hidden_layer', hidden_layer)
        model.add_module('output_layer', output_layer)
        # Load weights
        for name, param in model.named_parameters():
            if name in weights:
                param.data = torch.tensor(weights[name])
    else:
        # Load from model.pth
        model = torch.load('model.pth')
    return model

def create_dummy_image():
    # Create a simple image with digit 3
    img = Image.new('L', (28, 28), color='white')
    draw = ImageDraw.Draw(img)
    # Draw lines for digit 3
    draw.line([(10,10), (15,20), (20,10), (25,20), (30,10), (35,20), (40,10)], fill='black', width=2)
    draw.line([(10,20), (15,30), (20,20), (25,30), (30,20), (35,30), (40,20)], fill='black', width=2)
    draw.line([(10,30), (15,40), (20,30), (25,40), (30,30), (35,40), (40,30)], fill='black', width=2)
    return img

def predict(image_path, weights_path):
    model = load_model(weights_path)
    # Convert image to tensor
    img = Image.open(image_path).convert('L')
    img_array = np.array(img).astype(float) / 255.0
    img_tensor = torch.tensor(img_array, device='cpu').view(1, -1)
    output = model(img_tensor)
    pred = torch.argmax(output).item()
    print(pred)
    return pred

def main():
    if len(sys.argv) != 3:
        print("Usage: python3 infer.py <image> <weights>")
        sys.exit(1)
    image = sys.argv[1]
    weights = sys.argv[2]
    pred = predict(image, weights)
    with open('prediction.txt', 'w') as f:
        f.write(str(pred))

if __name__ == '__main__':
    main()
