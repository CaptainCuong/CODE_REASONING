import json
import requests

def create_spreadsheet(title):
    url = 'http://api:8000/spreadsheets/'
    data = {'title': title}
    response = requests.post(url, json=data)
    return response.json()['id']

def create_sheet(spreadsheet_id, title):
    url = f'http://api:8000/spreadsheets/{spreadsheet_id}/sheets/'
    data = {'title': title}
    response = requests.post(url, json=data)
    return response.json()['id']

def batch_update_cells(spreadsheet_id, sheet_id, data):
    # data is a list of rows, each row is a list of values for columns: Month, Revenue, Expenses, Profit
    # We'll map to cells with coordinates A1, B1, C1, D1 etc.
    updates = []
    for i, row in enumerate(data):
        for j, val in enumerate(row):
            col = 'A' if j==0 else 'B' if j==1 else 'C' if j==2 else 'D'
            cell_id = f'{col}{i+1}'
            updates.append({
                'value': val,
                'format': None
            })
    url = f'http://api:8000/sheets/{sheet_id}/cells/batch'
    data = {'updates': updates}
    response = requests.put(url, json=data)
    return response.json()

if __name__ == '__main__':
    spreadsheet_id = create_spreadsheet('Financial Report')
    sheet_id = create_sheet(spreadsheet_id, 'Q1 Data')
    data = [
        ['January', 10000, 8000, 2000],
        ['February', 12000, 9000, 3000],
        ['March', 15000, 10000, 5000]
    ]
    batch_update_cells(spreadsheet_id, sheet_id, data)
    print('Report created successfully')
EOF
