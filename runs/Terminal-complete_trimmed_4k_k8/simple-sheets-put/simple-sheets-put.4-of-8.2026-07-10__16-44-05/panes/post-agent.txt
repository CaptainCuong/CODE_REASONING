import openpyxl

# Create a new workbook
wb = openpyxl.Workbook()

# Create the Q1 Data sheet
wb.create_sheet('Q1 Data')

# Activate the sheet
ws = wb['Q1 Data']

# Write header
ws['A1'] = 'Month'
ws['B1'] = 'Revenue'
ws['C1'] = 'Expenses'

# Write data rows
data = [
    ('January', 10000, 8000),
    ('February', 12000, 9000),
    ('March', 15000, 10000)
]

for i, (month, revenue, expenses) in enumerate(data, start=2):
    ws[f'A{i}'] = month
    ws[f'B{i}'] = revenue
    ws[f'C{i}'] = expenses
    # Compute profit
    profit = revenue - expenses
    ws[f'D{i}'] = profit

# Save the workbook
wb.save('Financial Report.xlsx')

print('Report created: Financial Report.xlsx')
