import subprocess
import sys

def run_game(cmd):
    result = subprocess.run(['./ maze_game.sh'], capture_output=True, text=True)
    return result.stdout.strip()

def main():
    map_data = {}  # (x, y) -> {'open': True/False, 'exit': False}
    exit_pos = None
    current_pos = (0, 0)
    stack = []
    stack.append('N')
    while stack:
        direction = stack.pop()
        cmd = f'move {direction}'
        response = run_game(cmd)
        # print(f'{current_pos} -> {direction}: {response}')
        if response == 'reached exit':
            exit_pos = current_pos
            break
        elif response == 'moved':
            x, y = current_pos
            if direction == 'N':
                y += 1
            elif direction == 'E':
                x += 1
            elif direction == 'S':
                y -= 1
            elif direction == 'W':
                x -= 1
            current_pos = (x, y)
            map_data[(x, y)] = {'open': True, 'exit': False}
        elif response == 'hit wall':
            x, y = current_pos
            if direction == 'N':
                y += 1
            elif direction == 'E':
                x += 1
            elif direction == 'S':
                y -= 1
            elif direction == 'W':
                x -= 1
            map_data[(x, y)] = {'open': False, 'exit': False}
        else:
            print(f'Unknown response: {response}')
            sys.exit(1)

    if exit_pos:
        map_data[exit_pos]['exit'] = True

    # Explore size: find max and min coordinates
    if map_data:
        max_x = max(p[0] for p in map_data.keys())
        max_y = max(p[1] for p in map_data.keys())
        min_x = min(p[0] for p in map_data.keys())
        min_y = min(p[1] for p in map_data.keys())
    else:
        max_x = max_y = min_x = min_y = 0

    # Fill missing positions (if any) with '#'
    for y in range(min_y, max_y+1):
        for x in range(min_x, max_x+1):
            if (x, y) not in map_data:
                map_data[(x, y)] = {'open': False, 'exit': False}

    # Write map to file
    with open('/app/maze_map.txt', 'w') as f:
        for y in range(min_y, max_y+1):

                        line += ' '
                    elif map_data[(x, y)]['exit']:
                        line += 'E'
                    else:
                        line += '#'
                f.write(line + '\n')
