#!/usr/bin/env python3
import os
import sys
import shutil
import subprocess

def main():
    # Paths
    c4_reshard = '/app/c4_reshard'
    c4_sample = '/app/c4_sample'
    archive = os.path.join(c4_reshard, 'archive.tar.gz')

    # Ensure we have the archive file
    if not os.path.exists(archive):
        print('ERROR: archive.tar.gz not found')
        sys.exit(1)

    # Ensure we have exactly 30 parts
    parts = [f for f in os.listdir(c4_reshard) if f.startswith('archive.tar.gz.')]
    if len(parts) != 30:
        print(f'ERROR: Expected 30 parts, found {len(parts)}')
        sys.exit(1)

    # Join parts to recreate archive
    with open(archive, 'wb') as out:
        for part in parts:
            with open(os.path.join(c4_reshard, part), 'rb') as f:
                out.write(f.read())
