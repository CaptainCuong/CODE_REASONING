import numpy as np
import json
from scipy import optimize, signal

def fit_peak(x, y, peak_name):
    """Fit a single peak."""
    # Use a simple least squares fit of a function
    # Function: y = A * exp(- (x - x0)^2 / (2 * gamma^2)) + offset
    # We'll use scipy's optimize.curvefit.
    def f(x, x0, gamma, amp, offset):
        return amp * np.exp(- (x - x0)**2 / (2 * gamma**2)) + offset
    # We'll guess initial parameters.
    # For G peak, likely around 1600? Let's assume we know.
    # We'll let the algorithm find.
    # We'll use the first data point's x as x0 initial.
    popt, pcov = optimize.curvefit(f, x, y, p0=[x[0], 5, 1, 0.5])
    return {
        'x0': popt[0],
        'gamma': popt[1],
        'amplitude': popt[2],
        'offset': popt[3]
    }

def main():
    # Load data
    data = np.loadtxt('graphene.dat', delimiter=",")
    x = data[:, 0]
    y = data[:, 1]
    # Find indices of maxima
    max_idx, _ = signal.findpeaks(y)
    # We expect two peaks: G and 2D.
    # Let's take the first two maxima.
    peak_indices = max_idx[:2]
    results = {}
    for idx in peak_indices:
        # Determine which peak it is by x value.
        x_val = x[idx]
        if 1500 < x_val < 1700:
            peak_name = 'G'
        elif 2500 < x_val < 2900:
            peak_name = '2D'
        else:
            # If it's not one of the two, skip (should not happen)
            continue
        # Extract the peak segment: from the start of the peak to the end.
        # We'll find the nearest index with y > threshold before and after.
        # For simplicity, we'll take from idx-10 to idx+10.
        start = max(0, idx-10)
        end = min(len(x), idx+10)
        x_sub = x[start:end]
        y_sub = y[start:end]
        # Fit
        popt, pcov = optimize.curvefit(lambda x, x0, gamma, amp, offset: amp * np.exp(- (x - x0)**2 / (2 * gamma**2)) + offset, x_sub, y_sub, p0=[x_sub[0], 5, 1
, 0.5])
        results[peak_name] = {
            'x0': popt[0],
            'gamma': popt[1],
            'amplitude': popt[2],
            'offset': popt[3]
        }
    # Write to file
    with open('results.json', 'w') as f:
        json.dump(results, f, indent=2)
    print('Results written to results.json')

if __name__ == '__main__':
    main()
