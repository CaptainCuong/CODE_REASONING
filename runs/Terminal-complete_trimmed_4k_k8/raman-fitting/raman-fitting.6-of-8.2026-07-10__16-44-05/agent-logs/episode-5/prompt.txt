def fit_peak(x, y, peak_name):
    """Fit a single peak."""
    # Use a simple least squares fit of a function
    # Function: y = A * exp(- (x - x0)^2 / (2 * gamma^2)) + offset
    # We'll use scipy's optimize.curvefit.
    def f(x, x0, gamma, amp, offset):
        return amp * np.exp(- (x - x0)**2 / (2 * gamma**2)) + offset
    # We'll guess initial parameters.
    # For G peak, likely around 1600? Let's assume we know.
    # We'll let the algorithm find.
    # We'll use the first data point's x as x0 initial.
    popt, pcov = optimize.curvefit(f, x, y, p0=[x[0], 5, 1, 0.5])
    return {
        'x0': popt[0],

        'amplitude': popt[2],
        'offset': popt[3]
