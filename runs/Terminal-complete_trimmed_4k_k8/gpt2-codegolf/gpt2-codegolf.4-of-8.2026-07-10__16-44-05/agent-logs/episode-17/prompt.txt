import sys
import tensorflow as tf
import numpy as np
import os

def load_ckpt(filepath):
    # Load the Checkpoint
    ckpt = tf.train.load_checkpoint(filepath)
    # Convert to SavedModel
    tmpdir = '/tmp/savedmodel'
    tf.io.checkpoint_to_saved_model(ckpt, tmpdir)
    # Load SavedModel
    saved_model = tf.train.load_saved_model(os.path.join(tmpdir, 'saved_model.pb'))
    return saved_model

def main():
    # Load the model
    model = load_ckpt('gpt2-124M.ckpt')
    # Parse vocab.bpe
    with open('vocab.bpe', 'r') as f:
        lines = f.readlines()
    # Remove comments and empty lines
    vocab = []
    for line in lines:
        line = line.strip()
        if not line or line.startswith('#'):
            continue
        vocab.append(line)
    # Build mapping from string to id
    vocab_dict = {word: i for i, word in enumerate(vocab)}
    # Input string
    input_str = sys.argv[3]
    # Tokenize
    tokens = input_str.split()
    token_ids = []
    for token in tokens:
        if token in vocab_dict:
            token_ids.append(vocab_dict[token])
        else:
            # If not in vocab, maybe use a special token like 0
            token_ids.append(0)
    # Convert to tensor
    input_ids = tf.constant([token_ids], dtype=tf.int64)
    # Run inference
    output = model.predict(input_ids)
    # Get next 20 tokens
    next_tokens = output[0][:20]
    # Convert to strings
    next_strings = [vocab[t] for t in next_tokens]
    print(' '.join(next_strings))

if __name__ == '__main__':
    main()
