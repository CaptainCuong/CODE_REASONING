import sys
import tensorflow as tf

def load_vocabulary(filepath):
    with open(filepath, 'r') as f:
        lines = f.readlines()
    # skip version line
    tokens = []
    for line in lines[1:]:
        line = line.strip()
        if line:
            tokens.append(line)
    return tokens

def tokenize_string(s, tokens):
    n = len(s)
    token_ids = []
    i = 0
    while i < n:
        # try bigram
        if i+1 < n:
            bigram = s[i:i+2]
            for j in range(len(tokens)):
                if tokens[j] == bigram:
                    token_ids.append(j)
                    i += 1
                    break
            else:
                # try single character
                for j in range(len(tokens)):
                    if tokens[j] == s[i]:
                        token_ids.append(j)
                        i += 1
                        break
                else:
                    token_ids.append(0)
                    i += 1
        else:
            for j in range(len(tokens)):
                if tokens[j] == s[i]:
                    token_ids.append(j)
                    i += 1
                    break
            else:
                token_ids.append(0)
                i += 1
    return token_ids

def main():
    if len(sys.argv) != 4:
        print("Usage: python3 gpt2.py <ckpt> <bpe> <input>")
        sys.exit(1)
    ckpt = sys.argv[1]
    bpe = sys.argv[2]
    input_str = sys.argv[3]
    tokens = load_vocabulary(bpe)
    token_ids = tokenize_string(input_str, tokens)
    # Prepare input tensor
    input_tensor = tf.constant([token_ids], dtype=tf.int32)
    # Load model
    with tf.Graph().as_default():
        tf.load_graph(ckpt, None, None, None, None)
        op = tf.get_operations()[0]
        # Run inference
        result = tf.run(op, feed_dict={'input': input_tensor}, output_dict={'output': op.outputs[0]})
        # Extract token IDs and map to tokens
        ids = result['output'].astype(int).tolist()
        for i in range(min(20, len(ids))):
            print(tokens[ids[i]], end=' ')
        print()

if __name__ == "__main__":
    main()
