def explore_maze(maze_id):
    """Load and explore the given maze."""
    # Determine the path to the maze file
    # The server uses 'mazes' relative to its location, which is /app/../
    maze_dir = "/app/../mazes"
    if not os.path.exists(maze_dir):
        print(f"Maze directory {maze_dir} does not exist.")
        return None

    # Create a temporary copy to avoid overwriting original
    with tempfile.NamedTemporaryFile(mode='w', suffix='.txt', delete=False) as f:
        # Read the original file
        with open(os.path.join(maze_dir, f"maze_{ maze_id }.txt"), 'r') as orig:
            for line in orig:
                f.write(line)
        temp_path = f.name

    try:
        maze_data = read_maze_data(temp_path)
        game = MazeGame(maze_data)
        # Explore using DFS
        directions = list(game.directions.keys())
        visited_order = []
        for _ in range(100):
            for d in directions:
                result = game.move(d)
                if result == "moved" or result == "reached exit":
                    visited_order.append(d)
        # After exploration, the game's visited set should contain all positions
        # Build map
        map_str = "\n".join(["".join(row) for row in game.maze])
        # Remove the visited marker (if any) - but we don't need to
        # The map is already the original data.
        return map_str
    except Exception as e:
        print(f"Error exploring maze { maze_id }:{e}")
        return None
    finally:
        os.unlink(temp_path)
