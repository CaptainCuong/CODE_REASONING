> from http.server import HTTPServer, SimpleHTTPRequestHandler

> class FibonacciServer(SimpleHTTPRequestHandler):
>     def do_GET(self):
>         # Parse query parameters
>         query = self.path.split('?')[1] if '?' in self.path else ''
>         if not query:
>             self.send_error(400, 'Missing query parameter n')
>             return
>         # Split by '='
>         key, value = query.split('=', 1)
>         if key != 'n':
>             self.send_error(400, 'Query parameter n expected')
>             return
>         try:
>             n = int(value)
>         except ValueError:
>             self.send_error(400, 'Query parameter n must be an integer')
>             return
>         # Compute Fibonacci
>         def fibonacci(n):
>             if n < 0:
>                 return 0
>             a, b = 0, 1
>             for _ in range(n):
>                 a, b = b, a + b
>             return a
>         result = fibonacci(n)
>         response = {'result': result}
>         self.send_response(200)
>         self.send_header('Content-Type', 'application/json')
>         self.end_headers()
>         self.wfile.write(json.dumps(response))
>
> def main():
>     server = HTTPServer(("0.0.0.0", 3000), FibonacciServer)
>     print("Server started on port 3000")
>     server.serve_forever()

>     main()

import urllib.request
import urllib.error
import json

def test_missing_n():
    url = 'http://localhost:3000/fib'
    try:
        resp = urllib.request.urlopen(url)
        print('ERROR: Expected 400, got 200')
        resp.read()
    except urllib.error.HTTPError as e:
        if e.code == 400:
            print('OK: Missing n returns 400')
        else:
            print('ERROR: Missing n returns', e)
    except Exception as e:
        print('ERROR:', e)

def test_non_integer_n():
    url = 'http://localhost:3000/fib?n=abc'
    try:
        resp = urllib.request.urlopen(url)
        print('ERROR: Expected 400, got 200')
        resp.read()
    except urllib.error.HTTPError as e:
        if e.code == 400:
            print('OK: Non-integer n returns 400')
        else:
            print('ERROR: Non-integer n returns', e)
    except Exception as e:
        print('ERROR:', e)

def test_valid_integer_n():
    url = 'http://localhost:3000/fib?n=5'
    try:
        resp = urllib.request.urlopen(url)
        data = resp.read()
        parsed = json.loads(data)
        if parsed['result'] == 5:
            print('OK: Valid integer n returns correct result')
        else:
            print('ERROR: Valid integer n returns', parsed)
    except urllib.error.HTTPError as e:
        print('ERROR: Valid integer n returns', e)
    except Exception as e:
        print('ERROR:', e)

if __name__ == '__main__':
    test_missing_n()
    test_non_integer_n()
    test_valid_integer_n()
