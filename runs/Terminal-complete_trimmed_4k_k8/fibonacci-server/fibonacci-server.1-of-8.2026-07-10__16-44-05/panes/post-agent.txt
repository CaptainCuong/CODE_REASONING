> import json

> import urllib.parse
> import os

> class FibonacciHandler(http.server.SimpleHTTPRequestHandler):
>     def __init__(self, *args, **kwargs):
>         super().__init__(*args, **kwargs)
>         # Override default handler
>         self.default_path = '/fib'
>
>     def do_GET(self):
>         # Parse query parameters
>         query = urllib.parse.parse_qs(self.path.split('?', 1)[1] if '?' in self.path else '')
>         n_str = query.get('n')
>
>         if not n_str:
>             self.send_error(400, 'Missing query parameter n')
>             return
>
>         try:
>             n = int(n_str)
>             if n < 0:
>                 self.send_error(400, 'Negative integer')
>                 return
>             # Compute Fibonacci
>             def fib(m):
>                 if m <= 1:
>                     return 1
>                 return fib(m-1) + fib(m-2)
>             result = fib(n)
>             response = {'result': result}
>             self.write(json.dumps(response))
>             self.end_headers()
>         except ValueError:
>             self.send_error(400, 'Not an integer')
>             return
>
> def main():
>     handler = FibonacciHandler
>     http.server.test(port=3000)
>
> if __name__ == '__main__':
>     main()
> EOF
