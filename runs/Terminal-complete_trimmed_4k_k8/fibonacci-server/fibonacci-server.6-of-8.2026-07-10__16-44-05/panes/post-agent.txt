import http.server
import json
import sys

def fibonacci(n):
    if n < 0:
        raise ValueError('Negative index')
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

class FibonacciHandler(http.server.SimpleHTTPRequestHandler):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.protocol_version = 'HTTP/1.1'
        self.server_version = '1.0'

    def do_GET(self):
        print('do_GET called')
        query = self.path.split('?')[1] if '?' in self.path else ''
        params = {}
        for pair in query.split('&'):
            if '=' in pair:
                key, val = pair.split('=', 1)
                params[key] = val
        if 'n' not in params:
            self.send_response(400)
            self.end_headers()
            self.wfile.write(b'Query parameter \"n\" is missing')
            return
        try:
            n = int(params['n'])
        except ValueError:
            self.send_response(400)
            self.end_headers()
            self.wfile.write(b'Query parameter \"n\" must be an integer')
            return
        result = fibonacci(n)
        self.send_response(200)
        self.end_headers()
        self.wfile.write(json.dumps({'result': result}))

def main():
    port = 3000
    handler = FibonacciHandler
    httpd = http.server.HTTPServer(('0.0.0.0', port), handler)
    print('Server started')
    httpd.serve_forever()

if __name__ == '__main__':
    main()
