> import json

> import urllib.parse
> from http.server import SimpleHTTPRequestHandler, HTTPServer
>
> class FibonacciServer(SimpleHTTPRequestHandler):
>     def do_GET(self):
>         # Parse query string
>         query = urllib.parse.urlparse(self.path).query
>         params = urllib.parse.parse_qs(query)
>
>         # Check if 'n' parameter exists
>         if 'n' not in params:
>             self.send_error(400, 'Missing query parameter n')
>             return
>
>         # Extract and validate integer
>         try:
>             n = int(params['n'][0])
>         except ValueError:
>             self.send_error(400, 'Query parameter n must be an integer')
>             return
>
>         # Calculate Fibonacci
>         def fibonacci(n):
>             a, b = 0, 1
>             for _ in range(n):
>                 a, b = b, a + b
>             return a
>
>         result = fibonacci(n)
>         self.wfile.write(json.dumps({"result": result}).encode())
>         return
>
> if __name__ == '__main__':
>     # Bind to port 3000
>     handler = FibonacciServer
>     server = HTTPServer(('0.0.0.0', 3000), handler)
>     print(f'Serving on port 3000')
>     server.serve_forever()
