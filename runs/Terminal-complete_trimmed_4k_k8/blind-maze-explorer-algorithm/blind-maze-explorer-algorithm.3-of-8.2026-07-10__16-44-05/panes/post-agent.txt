import subprocess
import sys
import os

def explore(maze_id):
    # Run the game
    proc = subprocess.Popen(['./maze_game.sh', str(maze_id)],
                          stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
    # We'll explore and map
    # We'll keep a stack for DFS
    visited = set()
    map_data = {}
    # We'll also need to know the exit position
    exit_pos = None
    # We'll explore by sending batch commands to get a sense of size.
    # First, let's get a rough size by moving in directions until we hit a wall.
    # We'll do a simple while loop.
    # We'll start at (0,0) and explore.
    # We'll keep a set of visited positions.
    # We'll also need to know the size of the map; we can infer from the number of cells we visit.
    # But we need to know the size to write the map correctly.
    # We'll assume the map is small and we can explore until we have a complete map.
    # We'll use a while loop that continues until we have a complete map.
    # We'll also need to handle the fact that the terminal is interactive.
    # We'll send a batch of moves and parse the output.
    def send_batch(batch):
        proc.stdin.write(batch + '\n')
        output = proc.stdout.readline().strip()
        return output
    # We'll explore in a while loop.
    while True:
        # We'll try to move in each direction, but we need to avoid infinite loops.
        # We'll do a DFS.
        # We'll keep a stack of positions.
        stack = [(0,0)]
        visited = set()
        map_data = {}
        exit_pos = None
        # We'll explore until stack is empty.
        while stack:
            pos = stack.pop()
            visited.add(pos)
            map_data[pos] = ' '
            # Try to move in each direction
            for direction in ['N', 'E', 'S', 'W']:
                batch = f'move {direction}'
                out = send_batch(batch)
                if out == 'moved':
                    new_pos = pos
                    if direction == 'N':
                        new_pos = (pos[0]-1, pos[1])
                    elif direction == 'E':
                        new_pos = (pos[0], pos[1]+1)
                    elif direction == 'S':
                        new_pos = (pos[0]+1, pos[1])
                    elif direction == 'W':
                        new_pos = (pos[0], pos[1]-1)
                    if new_pos not in visited:
                        stack.append(new_pos)
                        map_data[new_pos] = ' '
                elif out == 'reached exit':
                    # We need to update the map to show 'E' for the exit position.
                    # We'll set the cell to 'E'.
                    map_data[new_pos] = 'E'
                    exit_pos = new_pos
                    break  # break out of direction loop
                # If hit wall, we ignore.
        # After exploring, we need to write the map.
        # We need to know the size of the map: find min and max coordinates.
        if map_data:
            xs = [p[0] for p in map_data]
            ys = [p[1] for p in map_data]
            min_x = min(xs)
            max_x = max(xs)
            min_y = min(ys)
            max_y = max(ys)
            # We'll create a string of lines.
            lines = []
            for i in range(min_y, max_y+1):
                line = ''
                for j in range(min_x, max_x+1):
                    line += map_data[(j, i)]
                lines.append(line)
            # Write to file.
            with open(f'/app/output/{maze_id}.txt', 'w') as f:
                f.write('\n'.join(lines))
            print(f'Maze {maze_id} mapped.')
            return map_data, exit_pos
        else:
            print('No cells explored.')
            return None, None

if __name__ == '__main__':
    # Test on maze 1
    explore(1)
